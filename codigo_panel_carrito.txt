'use client'

import { useState } from 'react'

const inputStyle = {
  width: '100%',
  padding: '8px 10px',
  fontSize: '14px',
  borderRadius: '6px',
  border: '1px solid #ccc',
  marginBottom: '8px',
  boxSizing: 'border-box',
}

export default function EditarDireccionBoton({ pedido }) {
  const [abierto, setAbierto] = useState(false)
  const [form, setForm] = useState({
    calle: pedido.direccion?.calle || '',
    numero: pedido.direccion?.numero || '',
    pisoDepto: pedido.direccion?.pisoDepto || '',
    ciudad: pedido.direccion?.ciudad || '',
    codigoPostal: pedido.direccion?.codigoPostal || '',
  })
  const [cantidadBultos, setCantidadBultos] = useState(1)
  const [opciones, setOpciones] = useState([])
  const [opcionElegida, setOpcionElegida] = useState(null)
  const [sucursalElegida, setSucursalElegida] = useState(null)
  const [cotizando, setCotizando] = useState(false)
  const [guardando, setGuardando] = useState(false)
  const [mensaje, setMensaje] = useState('')

  function cambiar(campo, valor) {
    setForm((f) => ({ ...f, [campo]: valor }))
  }

  async function cotizar() {
    if (!form.codigoPostal) {
      setMensaje('Ingresá el código postal')
      return
    }
    setCotizando(true)
    setMensaje('')
    setOpciones([])
    setOpcionElegida(null)
    setSucursalElegida(null)
    try {
      const res = await fetch('/api/envios/cotizar', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ postalCode: form.codigoPostal, items: [{ cantidad: cantidadBultos }] }),
      })
      const data = await res.json()
      if (!data.success) {
        setMensaje(data.error || 'No se pudo cotizar')
        return
      }
      setOpciones(data.rates || [])
    } catch (e) {
      setMensaje('Error de conexión al cotizar')
    } finally {
      setCotizando(false)
    }
  }

  async function guardar() {
    if (!form.calle || !form.numero || !form.ciudad || !form.codigoPostal) {
      setMensaje('Completá la dirección')
      return
    }
    setGuardando(true)
    setMensaje('')
    try {
      const body = {
        pedidoId: pedido.id,
        direccion: form,
      }
      if (opcionElegida) {
        body.envio = {
          carrierSlug: opcionElegida.carrierSlug,
          serviceCode: opcionElegida.serviceCode,
          carrier: opcionElegida.carrier,
          service: opcionElegida.service,
          price: opcionElegida.price,
          branchCode: sucursalElegida?.branchCode || null,
        }
      }
      const res = await fetch('/api/admin/pedidos-editar-direccion', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
      })
      const data = await res.json()
      if (!res.ok) {
        setMensaje(data.error || 'No se pudo guardar')
        return
      }
      setMensaje('Guardado. Recargá la página para ver los cambios.')
    } catch (e) {
      setMensaje('Error de conexión al guardar')
    } finally {
      setGuardando(false)
    }
  }

  if (!abierto) {
    return (
      <button
        onClick={() => setAbierto(true)}
        style={{
          marginTop: '10px',
          marginRight: '8px',
          padding: '8px 14px',
          fontSize: '13px',
          fontWeight: '600',
          borderRadius: '8px',
          border: '1px solid #666',
          backgroundColor: '#fff',
          color: '#333',
          cursor: 'pointer',
        }}
      >
        ✏️ Editar dirección
      </button>
    )
  }

  return (
    <div style={{ marginTop: '12px', padding: '14px', borderRadius: '8px', border: '1px solid #ddd', backgroundColor: '#fafafa' }}>
      <div style={{ display: 'flex', gap: '8px' }}>
        <div style={{ flex: 2 }}>
          <input style={inputStyle} placeholder="Calle" value={form.calle} onChange={(e) => cambiar('calle', e.target.value)} />
        </div>
        <div style={{ flex: 1 }}>
          <input style={inputStyle} placeholder="Número" value={form.numero} onChange={(e) => cambiar('numero', e.target.value)} />
        </div>
      </div>
      <input style={inputStyle} placeholder="Piso / Depto" value={form.pisoDepto} onChange={(e) => cambiar('pisoDepto', e.target.value)} />
      <div style={{ display: 'flex', gap: '8px' }}>
        <div style={{ flex: 2 }}>
          <input style={inputStyle} placeholder="Ciudad" value={form.ciudad} onChange={(e) => cambiar('ciudad', e.target.value)} />
        </div>
        <div style={{ flex: 1 }}>
          <input style={inputStyle} placeholder="Código postal" value={form.codigoPostal} onChange={(e) => cambiar('codigoPostal', e.target.value)} />
        </div>
      </div>

      <div style={{ display: 'flex', gap: '8px', alignItems: 'center', marginBottom: '8px' }}>
        <input
          type="number"
          min="1"
          style={{ ...inputStyle, marginBottom: 0, width: '80px' }}
          value={cantidadBultos}
          onChange={(e) => setCantidadBultos(e.target.value)}
        />
        <button
          onClick={cotizar}
          disabled={cotizando}
          style={{
            padding: '8px 12px',
            fontSize: '13px',
            borderRadius: '6px',
            border: 'none',
            backgroundColor: '#009ee3',
            color: '#fff',
            cursor: cotizando ? 'not-allowed' : 'pointer',
          }}
        >
          {cotizando ? 'Cotizando...' : 'Volver a cotizar envío'}
        </button>
      </div>

      {opciones.length > 0 && (
        <div style={{ marginBottom: '10px' }}>
          {opciones.map((op) => (
            <div
              key={op.id}
              onClick={() => { setOpcionElegida(op); setSucursalElegida(null) }}
              style={{
                padding: '8px 10px',
                borderRadius: '6px',
                border: opcionElegida?.id === op.id ? '2px solid #009ee3' : '1px solid #ddd',
                marginBottom: '6px',
                cursor: 'pointer',
                fontSize: '13px',
              }}
            >
              <strong>{op.carrier} — {op.service}</strong> · ${op.price}
            </div>
          ))}

          {opcionElegida?.branches?.length > 0 && (
            <select
              style={inputStyle}
              value={sucursalElegida?.branchCode || ''}
              onChange={(e) => {
                const suc = opcionElegida.branches.find((b) => b.branchCode === e.target.value)
                setSucursalElegida(suc || null)
              }}
            >
              <option value="">Elegí una sucursal</option>
              {opcionElegida.branches.map((b) => (
                <option key={b.branchCode} value={b.branchCode}>{b.direccion}</option>
              ))}
            </select>
          )}
        </div>
      )}

      {mensaje && <p style={{ fontSize: '13px', color: '#b91c1c', marginBottom: '8px' }}>{mensaje}</p>}

      <div style={{ display: 'flex', gap: '8px' }}>
        <button
          onClick={guardar}
          disabled={guardando}
          style={{
            padding: '8px 14px',
            fontSize: '13px',
            fontWeight: '600',
            borderRadius: '6px',
            border: 'none',
            backgroundColor: '#15803d',
            color: '#fff',
            cursor: guardando ? 'not-allowed' : 'pointer',
          }}
        >
          {guardando ? 'Guardando...' : 'Guardar'}
        </button>
        <button
          onClick={() => setAbierto(false)}
          style={{
            padding: '8px 14px',
            fontSize: '13px',
            borderRadius: '6px',
            border: '1px solid #ccc',
            backgroundColor: '#fff',
            cursor: 'pointer',
          }}
        >
          Cancelar
        </button>
      </div>
    </div>
  )
}

// DESTINO: src/app/api/crear-pedido-transferencia/route.js  (reemplaza el archivo actual)
import { NextResponse } from 'next/server'
import { createClient } from '@supabase/supabase-js'
import { DESCUENTO_TRANSFERENCIA } from '@/lib/configTransferencia'
import { calcularCostoEnvio } from '@/lib/configEnvioGratis'
import { nombreLineaCupon } from '@/lib/configCupones'
import {
  validarCuponParaPedido,
  usarCupon,
  devolverUsoCupon,
  usuarioDesdeRequest,
  reservarUsoCuenta,
  liberarUsoCuenta,
  vincularUsoConPedido,
  MENSAJE_LOGIN,
  MENSAJE_YA_USADO,
} from '@/lib/cupones'
import {
  enviarEmailInstruccionesTransferencia,
  enviarEmailNotificacionTransferencia,
} from '@/lib/enviarEmails'

const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY
)

async function verificarRecaptcha(token) {
  if (!token) return false

  try {
    const res = await fetch('https://www.google.com/recaptcha/api/siteverify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: `secret=${process.env.RECAPTCHA_SECRET_KEY}&response=${token}`,
    })

    const data = await res.json()
    return data.success && data.score >= 0.5 && data.action === 'checkout'
  } catch (error) {
    console.error('Error al verificar reCAPTCHA:', error)
    return false
  }
}

export async function POST(request) {
  // Si el pedido no se llega a guardar, se devuelven los usos que se habían reservado
  let cuponReservado = null
  let usoCuentaReservado = null // { userId, codigo }

  async function deshacerReservas() {
    if (cuponReservado) {
      await devolverUsoCupon(cuponReservado)
      cuponReservado = null
    }
    if (usoCuentaReservado) {
      await liberarUsoCuenta(usoCuentaReservado.userId, usoCuentaReservado.codigo)
      usoCuentaReservado = null
    }
  }

  try {
    const { items, zonaEnvio, datosCliente, recaptchaToken, cuponCodigo } = await request.json()

    const recaptchaValido = await verificarRecaptcha(recaptchaToken)
    if (!recaptchaValido) {
      return NextResponse.json(
        { error: 'No pudimos verificar que sos una persona real. Recargá la página e intentá de nuevo.' },
        { status: 400 }
      )
    }

    if (!Array.isArray(items) || items.length === 0) {
      return NextResponse.json({ error: 'Tu carrito está vacío' }, { status: 400 })
    }
    if (!zonaEnvio) {
      return NextResponse.json({ error: 'Elegí una zona de envío' }, { status: 400 })
    }
    if (!datosCliente?.nombre || !datosCliente?.email) {
      return NextResponse.json({ error: 'Faltan tus datos de contacto' }, { status: 400 })
    }

    // Mismo criterio que crear-preferencia: se toman nombre y precio del carrito.
    // Se limpian cantidad y precio para que no entren valores raros.
    const itemsPedido = items.map((item) => {
      const cantidad = Math.max(1, parseInt(item.cantidad, 10) || 1)
      const precio = Math.max(0, Number(item.precio) || 0)
      const detalles = [item.color, item.talle ? `Talle ${item.talle}` : null].filter(Boolean).join(' - ')

      return {
        nombre: detalles ? `${item.nombre} (${detalles})` : item.nombre,
        cantidad,
        precio,
        imagen_url: item.imagen_url || null,
      }
    })

    const subtotal = itemsPedido.reduce((acc, item) => acc + item.precio * item.cantidad, 0)

    // El cupón y el descuento por transferencia SE SUMAN: primero se descuenta el cupón
    // y el % de transferencia se calcula sobre lo que queda (5% + 5% = 9,75%).
    let cupon = null
    let descuentoCupon = 0

    if (cuponCodigo) {
      // Para usar un cupón hay que haber ingresado con Google
      const usuario = await usuarioDesdeRequest(request)
      if (!usuario) {
        return NextResponse.json(
          { error: MENSAJE_LOGIN, cupon_invalido: true, login_requerido: true },
          { status: 401 }
        )
      }

      // El cupón se valida y se calcula acá, nunca se confía en el descuento que manda el navegador
      const validacion = await validarCuponParaPedido(cuponCodigo, subtotal, usuario.id)
      if (!validacion.ok) {
        return NextResponse.json({ error: validacion.error, cupon_invalido: true }, { status: 400 })
      }

      const reservado = await usarCupon(validacion.cupon.codigo)
      if (!reservado) {
        return NextResponse.json(
          { error: 'El cupón no es válido o ya no está disponible.', cupon_invalido: true },
          { status: 400 }
        )
      }
      cuponReservado = validacion.cupon.codigo

      const uso = await reservarUsoCuenta(usuario.id, validacion.cupon.codigo)
      if (uso !== 'ok') {
        await deshacerReservas()
        return NextResponse.json(
          {
            error: uso === 'ya_usado' ? MENSAJE_YA_USADO : 'No pudimos verificar el cupón. Probá de nuevo.',
            cupon_invalido: true,
          },
          { status: 400 }
        )
      }
      usoCuentaReservado = { userId: usuario.id, codigo: validacion.cupon.codigo }

      cupon = validacion.cupon
      descuentoCupon = validacion.descuento
    }

    const subtotalConCupon = subtotal - descuentoCupon
    const descuento = Math.round(subtotalConCupon * DESCUENTO_TRANSFERENCIA)
    const costoEnvio = calcularCostoEnvio(subtotalConCupon, zonaEnvio)
    const total = subtotalConCupon - descuento + costoEnvio

    // Los descuentos quedan como líneas con precio negativo, así todo suma igual
    // en el admin, en los emails y en la factura
    if (descuentoCupon > 0) {
      itemsPedido.push({
        nombre: nombreLineaCupon(cupon),
        cantidad: 1,
        precio: -descuentoCupon,
        imagen_url: null,
      })
    }

    if (descuento > 0) {
      itemsPedido.push({
        nombre: `Descuento por transferencia (${Math.round(DESCUENTO_TRANSFERENCIA * 100)}%)`,
        cantidad: 1,
        precio: -descuento,
        imagen_url: null,
      })
    }

    const direccion = datosCliente.calle
      ? {
          calle: datosCliente.calle,
          numero: datosCliente.numero,
          pisoDepto: datosCliente.pisoDepto || null,
          ciudad: datosCliente.ciudad,
          codigoPostal: datosCliente.codigoPostal || null,
        }
      : null

    // La tabla identifica cada pedido por esta columna: se usa una referencia propia que arranca con TR-
    const referencia =
      'TR-' + Date.now().toString(36).toUpperCase() + Math.random().toString(36).slice(2, 6).toUpperCase()

    const { data: pedido, error: errorInsert } = await supabaseAdmin
      .from('pedidos')
      .insert({
        mercadopago_payment_id: referencia,
        estado: 'pending',
        metodo_pago: 'transferencia',
        descuento,
        cupon_codigo: cupon ? cupon.codigo : null,
        descuento_cupon: descuentoCupon,
        total,
        items: itemsPedido,
        comprador_email: datosCliente.email || null,
        cliente_nombre: datosCliente.nombre || null,
        cliente_dni: datosCliente.dni || null,
        cliente_telefono: datosCliente.telefono || null,
        direccion,
        zona_envio: zonaEnvio.nombre || null,
        costo_envio: costoEnvio,
        envio_carrier: zonaEnvio.envioCarrier || zonaEnvio.envio_carrier || null,
        envio_service_code: zonaEnvio.envioServiceCode || zonaEnvio.envio_service_code || null,
        envio_branch_code: zonaEnvio.envioBranchCode || zonaEnvio.envio_branch_code || null,
        notas: datosCliente.notas || null,
      })
      .select('id')
      .single()

    if (errorInsert) {
      console.error('Error al guardar el pedido por transferencia:', errorInsert)
      await deshacerReservas()
      return NextResponse.json({ error: 'No se pudo guardar el pedido. Probá de nuevo.' }, { status: 500 })
    }

    // El pedido ya está guardado: los usos del cupón quedan firmes aunque falle algo después (por ejemplo un email)
    if (usoCuentaReservado) {
      await vincularUsoConPedido(usoCuentaReservado.userId, usoCuentaReservado.codigo, pedido.id)
    }
    cuponReservado = null
    usoCuentaReservado = null

    const pedidoParaEmail = {
      id: pedido.id,
      cliente_nombre: datosCliente.nombre || null,
      comprador_email: datosCliente.email || null,
      cliente_telefono: datosCliente.telefono || null,
      cliente_dni: datosCliente.dni || null,
      items: itemsPedido,
      total,
      zona_envio: zonaEnvio.nombre || null,
      costo_envio: costoEnvio,
      direccion,
      notas: datosCliente.notas || null,
    }

    await enviarEmailInstruccionesTransferencia(pedidoParaEmail)
    await enviarEmailNotificacionTransferencia(pedidoParaEmail)

    return NextResponse.json({ pedido_id: pedido.id, total, descuento, descuento_cupon: descuentoCupon })
  } catch (error) {
    console.error('Error al crear pedido por transferencia:', error)
    await deshacerReservas()
    return NextResponse.json({ error: 'Error al crear el pedido' }, { status: 500 })
  }
}

// DESTINO: src/app/api/crear-preferencia/route.js  (reemplaza el archivo actual)
import { MercadoPagoConfig, Preference } from 'mercadopago'
import { NextResponse } from 'next/server'
import { calcularCostoEnvio } from '@/lib/configEnvioGratis'
import {
  validarCuponParaPedido,
  usarCupon,
  devolverUsoCupon,
  repartirDescuentoEnItems,
  usuarioDesdeRequest,
  MENSAJE_LOGIN,
} from '@/lib/cupones'

const client = new MercadoPagoConfig({
  accessToken: process.env.MERCADOPAGO_ACCESS_TOKEN,
})

async function verificarRecaptcha(token) {
  if (!token) return false

  try {
    const res = await fetch('https://www.google.com/recaptcha/api/siteverify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: `secret=${process.env.RECAPTCHA_SECRET_KEY}&response=${token}`,
    })

    const data = await res.json()

    // score va de 0 (bot) a 1 (humano). 0.5 es el umbral recomendado por Google.
    return data.success && data.score >= 0.5 && data.action === 'checkout'
  } catch (error) {
    console.error('Error al verificar reCAPTCHA:', error)
    return false
  }
}

export async function POST(request) {
  // Si el pago no se llega a crear, se devuelve el uso del cupón que se había reservado
  let cuponReservado = null

  try {
    const { items, zonaEnvio, datosCliente, recaptchaToken, cuponCodigo } = await request.json()

    const recaptchaValido = await verificarRecaptcha(recaptchaToken)
    if (!recaptchaValido) {
      return NextResponse.json(
        { error: 'No pudimos verificar que sos una persona real. Recargá la página e intentá de nuevo.' },
        { status: 400 }
      )
    }

    const subtotalProductos = items.reduce((acc, item) => acc + item.precio * item.cantidad, 0)

    // Cupón: se valida y se calcula acá, nunca se confía en el descuento que manda el navegador.
    // Para usar un cupón hay que haber ingresado con Google.
    let cupon = null
    let descuentoCupon = 0
    let usuario = null

    if (cuponCodigo) {
      usuario = await usuarioDesdeRequest(request)
      if (!usuario) {
        return NextResponse.json(
          { error: MENSAJE_LOGIN, cupon_invalido: true, login_requerido: true },
          { status: 401 }
        )
      }

      const validacion = await validarCuponParaPedido(cuponCodigo, subtotalProductos, usuario.id)
      if (!validacion.ok) {
        return NextResponse.json({ error: validacion.error, cupon_invalido: true }, { status: 400 })
      }
      if (validacion.descuento >= subtotalProductos) {
        return NextResponse.json(
          { error: 'Este cupón cubre todo el pedido y no se puede usar con Mercado Pago. Escribinos y lo resolvemos.', cupon_invalido: true },
          { status: 400 }
        )
      }

      const reservado = await usarCupon(validacion.cupon.codigo)
      if (!reservado) {
        return NextResponse.json(
          { error: 'El cupón no es válido o ya no está disponible.', cupon_invalido: true },
          { status: 400 }
        )
      }

      cuponReservado = validacion.cupon.codigo
      cupon = validacion.cupon
      descuentoCupon = validacion.descuento
    }

    const preference = new Preference(client)

    const itemsParaMP = items.map((item) => {
      const detalles = [item.color, item.talle ? `Talle ${item.talle}` : null]
        .filter(Boolean)
        .join(' - ')

      return {
        id: String(item.id),
        title: detalles ? `${item.nombre} (${detalles})` : item.nombre,
        quantity: item.cantidad,
        unit_price: item.precio,
        currency_id: 'ARS',
      }
    })

    // Mercado Pago no acepta líneas con precio negativo: el descuento del cupón se reparte
    // entre los precios de los productos (el total queda correcto)
    const itemsConCupon = repartirDescuentoEnItems(itemsParaMP, descuentoCupon)

    // Envío gratis desde cierto monto (medido después del cupón): el costo se recalcula acá,
    // no se confía en el del navegador
    const subtotalConCupon = subtotalProductos - descuentoCupon
    const costoEnvio = calcularCostoEnvio(subtotalConCupon, zonaEnvio)

    if (zonaEnvio && costoEnvio > 0) {
      itemsConCupon.push({
        id: 'envio',
        title: `Envío - ${zonaEnvio.nombre}`,
        quantity: 1,
        unit_price: costoEnvio,
        currency_id: 'ARS',
      })
    }

    // Guardamos también los items originales (con imagen_url) en metadata,
    // porque Mercado Pago no permite campos custom dentro de "items".
    const itemsOriginales = items.map((item) => ({
      nombre: item.nombre,
      color: item.color || null,
      talle: item.talle || null,
      imagen_url: item.imagen_url || null,
    }))

    const resultado = await preference.create({
      body: {
        items: itemsConCupon,
        payer: {
          name: datosCliente?.nombre,
          email: datosCliente?.email,
        },
        metadata: {
          datos_cliente: datosCliente,
          zona_envio: zonaEnvio ? { ...zonaEnvio, costo: costoEnvio } : null,
          items_originales: itemsOriginales,
          cupon: cupon ? { codigo: cupon.codigo, descuento: descuentoCupon } : null,
          // Quién usó el cupón: al aprobarse el pago, el webhook lo anota en esa cuenta
          usuario_id: usuario ? usuario.id : null,
        },
        back_urls: {
          success: 'https://www.dimedetiambos.com.ar/pago-exitoso',
          failure: 'https://www.dimedetiambos.com.ar/pago-fallido',
          pending: 'https://www.dimedetiambos.com.ar/pago-pendiente',
        },
        auto_return: 'approved',
        notification_url: 'https://www.dimedetiambos.com.ar/api/webhook-mercadopago',
      },
    })

    return NextResponse.json({ id: resultado.id, init_point: resultado.init_point })
  } catch (error) {
    console.error('Error al crear preferencia:', error)
    if (cuponReservado) await devolverUsoCupon(cuponReservado)
    return NextResponse.json({ error: 'Error al crear la preferencia de pago' }, { status: 500 })
  }
}

  const checkoutIniciado = useRef(false)

  useEffect(() => {
    if (checkoutIniciado.current || items.length === 0) return
    checkoutIniciado.current = true
    track('InitiateCheckout', {
      value: totalProductos,
      currency: 'ARS',
      content_type: 'product',
      num_items: items.reduce((a, i) => a + i.cantidad, 0),

  // y el % de transferencia se calcula sobre lo que queda (5% + 5% = 9,75%).
  const hayCupon = cuponAplica && descuentoCupon > 0
  const cuponEnUso = hayCupon
  const descuentoCuponEfectivo = cuponEnUso ? descuentoCupon : 0
  const subtotalEfectivo = totalProductos - descuentoCuponEfectivo

// Orden del catálogo: primero ambos, luego chaquetas y al final pantalones.
// Se mira cómo EMPIEZA el nombre, para que "Ambo ... Pantalon con Puños" no caiga en pantalones.
function grupo(nombre = '') {
  const n = nombre
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .trim()

  if (n.startsWith('ambo')) return 0
  if (n.startsWith('chaqueta')) return 1
  if (n.startsWith('pantalon')) return 2
  return 3
}

// Fotos de TABLA DE TALLES que no deben aparecer nunca en la tarjeta del home.
// Pegá acá el link de cada foto de tabla, entre comillas y separados por coma. Ejemplo:
//   'https://http2.mlstatic.com/D_Q_NP_000000-MLA000000000_000000-F.webp',
const FOTOS_TABLA_TALLES = [
]

const CAMPOS_FOTO = [
  'imagen_url', 'imagen_2', 'imagen_3', 'imagen_4', 'imagen_5', 'imagen_6',
  'imagen_7', 'imagen_8', 'imagen_9', 'imagen_10', 'imagen_11', 'imagen_12',
]

// Fotos de una variante, sin las de tabla de talles
function fotosValidas(variante) {
  return CAMPOS_FOTO.map((campo) => variante[campo]).filter(
    (url) => url && !FOTOS_TABLA_TALLES.includes(url)
  )
}

export default function ProductoCardCliente({ producto, variantes }) {
  const coloresUnicos = Array.from(new Set(variantes.map((v) => v.color).filter(Boolean)))

  // Una foto por color: la primera que no sea tabla de talles
  const primeraFotoPorColor = coloresUnicos
    .map((color) => {
      for (const v of variantes.filter((x) => x.color === color)) {
        const url = fotosValidas(v)[0]
        if (url) return { ...v, imagen_url: url }
      }
      return null
    })
    .filter(Boolean)

  const fotoNegro = primeraFotoPorColor.find((f) => f.color?.toLowerCase() === 'negro')
  const fotoBlanco = primeraFotoPorColor.find((f) => f.color?.toLowerCase() === 'blanco')
  const fotoPorDefecto = fotoNegro || fotoBlanco || primeraFotoPorColor[0]

  // Miniaturas: Negro siempre primero (si el producto lo tiene); el resto mantiene su orden
  const esNegro = (f) => (f.color?.toLowerCase() === 'negro' ? 1 : 0)
  const fotosOrdenadas = [...primeraFotoPorColor].sort((a, b) => esNegro(b) - esNegro(a))

  const [fotoActiva, setFotoActiva] = useState(fotoPorDefecto)
  const [modalAbierto, setModalAbierto] = useState(false)
  const [colorSeleccionado, setColorSeleccionado] = useState(fotoActiva?.color || '')
  const [varianteElegida, setVarianteElegida] = useState(fotoPorDefecto || variantes[0] || null)
  const [cantidad, setCantidad] = useState(1)

  const { agregarAlCarrito, abrirPanel, cerrarPanel } = useCarrito()

  const variantesDelColor = variantes.filter((v) => v.color === colorSeleccionado)

  function abrirModal(e) {
    e.preventDefault()
    cerrarPanel()
    setColorSeleccionado(fotoActiva?.color || variantes[0]?.color || '')
    setVarianteElegida(fotoActiva || variantes[0] || null)
    setCantidad(1)
    setModalAbierto(true)
  }

  function handleSeleccionarColor(nuevoColor) {
    setColorSeleccionado(nuevoColor)
    const opcionesDelColor = variantes.filter((v) => v.color === nuevoColor)
    if (opcionesDelColor.length > 0) setVarianteElegida(opcionesDelColor[0])
  }

  function handleSeleccionarTalle(varianteId) {
    const encontrada = variantesDelColor.find((v) => String(v.id) === String(varianteId))
    if (encontrada) setVarianteElegida(encontrada)
  }

function handleAgregar() {
  agregarAlCarrito(
    {
      ...producto,
      variante_id: varianteElegida?.id,
      color: varianteElegida?.color,
      talle: varianteElegida?.talle,
      imagen_url: varianteElegida?.imagen_url,
    },
    cantidad
  )

  track('AddToCart', {
    value: producto.precio * cantidad,
    currency: 'ARS',
    content_type: 'product',
    content_ids: [String(producto.id)],
    content_name: producto.nombre,
    contents: [{ id: String(producto.id), quantity: cantidad, item_price: producto.precio }],
  })

  setModalAbierto(false)
  abrirPanel()
}

  return (
    // flex-col: la tarjeta se estira al alto de su fila y el botón queda siempre abajo (mt-auto)
    <div className="flex w-[250px] flex-col rounded-2xl border border-[#E4DCCF] bg-white overflow-hidden transition-shadow hover:shadow-[0_8px_24px_-8px_rgba(46,42,38,0.18)]">
      <Link href={`/producto/${producto.id}`} prefetch={false} className="block shrink-0">
        {/* Mismo recuadro para todas las fotos. Fondo blanco: los márgenes de la foto no se notan */}
        <div className="aspect-[4/5] w-full overflow-hidden bg-white">
          {fotoActiva?.imagen_url && (
            <img
              src={fotoActiva.imagen_url}
              alt={`${producto.nombre} - ${fotoActiva.color}`}
              loading="lazy"
              decoding="async"
              className="h-full w-full object-cover object-top"
            />
          )}
        </div>
      </Link>

      {/* Espacio de miniaturas SIEMPRE reservado (12px de padding + 36px de miniatura = 48px),
          así un producto de un solo color no desalinea el precio de las tarjetas vecinas */}
      <div className="flex min-h-[48px] gap-1.5 px-3 pt-3 flex-wrap">
        {primeraFotoPorColor.length > 1 &&

              onClick={(e) => {
                e.preventDefault()
                setFotoActiva(f)

              className={`h-9 w-9 overflow-hidden rounded-md border transition ${
                fotoActiva?.color === f.color
                  ? 'border-[#2F6B63] ring-1 ring-[#2F6B63]'
                  : 'border-[#E4DCCF]'
              }`}
              title={f.color}
            >
              <img src={f.imagen_url} alt={f.color} loading="lazy" decoding="async" className="h-full w-full object-cover" />
            </button>
          ))}
      </div>

      <Link href={`/producto/${producto.id}`} prefetch={false} className="block px-4 pt-3">
        {/* Reserva siempre 3 líneas: 3 x 1.3 = 3.9em. Así el precio queda a la misma altura */}
        <h2 className="font-serif text-[17px] leading-[1.3] min-h-[3.9em] text-[#2E2A26]">

                  {coloresUnicos.map((color) => (
                    <option key={color} value={color}>{color}</option>
                  ))}
                </select>
              </div>
            )}

            {variantesDelColor.length > 0 && (
              <div className="mt-4">
                <label className="mb-1.5 block text-sm font-medium text-[#2E2A26]">
                  Talle: <span className="font-normal text-[#8A8378]">{varianteElegida?.talle}</span>
                </label>
                <select
                  value={varianteElegida?.id || ''}
                  onChange={(e) => handleSeleccionarTalle(e.target.value)}
                  className="w-full rounded-lg border border-[#E4DCCF] bg-white px-3 py-2.5 text-sm text-[#2E2A26] outline-none focus:border-[#2F6B63]"
                >
                  {variantesDelColor.map((v) => (
                    <option key={v.id} value={v.id} disabled={v.stock === 0}>
                      {v.talle} {v.stock === 0 ? '(Sin stock)' : ''}
                    </option>
                  ))}
                </select>
              </div>
            )}

            {varianteElegida && varianteElegida.stock > 0 && (
              <div className="mt-4">
                <label className="mb-1.5 block text-sm font-medium text-[#2E2A26]">
                  Cantidad: <span className="font-normal text-[#8A8378]">({varianteElegida.stock} disponibles)</span>
                </label>
                <div className="flex items-center">
                  <button
                    onClick={() => setCantidad((c) => Math.max(1, c - 1))}
                    disabled={cantidad <= 1}
                    className="h-9 w-9 rounded-l-lg border border-[#E4DCCF] text-[#2E2A26] disabled:text-[#CFC7B8]"
                  >
                    −
                  </button>
                  <div className="flex h-9 w-11 items-center justify-center border-y border-[#E4DCCF] text-sm font-medium">
                    {cantidad}
                  </div>
                  <button
                    onClick={() => setCantidad((c) => Math.min(varianteElegida.stock, c + 1))}
                    disabled={cantidad >= varianteElegida.stock}
                    className="h-9 w-9 rounded-r-lg border border-[#E4DCCF] text-[#2E2A26] disabled:text-[#CFC7B8]"
                  >
                    +
                  </button>
                </div>
              </div>
            )}

            <div className="mt-5 rounded-xl border border-[#E4DCCF] bg-[#F3EEE6] px-3 py-2.5 text-sm text-[#2E2A26]">
              <p className="font-semibold text-[#2F6B63]">🧵 Confeccionado a pedido</p>
              <p className="mt-0.5 text-[#5C564C]">
                Tiempo de confección: <strong>{textoConfeccion()}</strong>
              </p>
            </div>

            <button
              onClick={handleAgregar}
              disabled={!varianteElegida || varianteElegida?.stock === 0}
              className="mt-4 w-full rounded-full bg-[#2F6B63] py-3 text-sm font-semibold text-white transition hover:bg-[#28584F] disabled:bg-[#E4DCCF] disabled:text-[#8A8378]"
            >
              {varianteElegida?.stock === 0 ? 'Sin stock disponible' : 'Agregar al carrito'}
            </button>
          </div>
        </div>,
        document.body
      )}
    </div>
  )
}

// DESTINO: src/lib/carrito.js  (reemplaza el archivo actual)
'use client'
import { createContext, useContext, useState, useEffect } from 'react'
import { UMBRAL_ENVIO_GRATIS, calcularCostoEnvio } from '@/lib/configEnvioGratis'
import { calcularDescuentoCupon, normalizarCodigo } from '@/lib/configCupones'

const CarritoContext = createContext()

const CLAVE_ZONA = 'dimedetiambos_zona_envio'
const CLAVE_CUPON = 'dimedetiambos_cupon'

export function CarritoProvider({ children }) {
  const [items, setItems] = useState([])
  const [zonaEnvio, setZonaEnvio] = useState(null)
  const [cupon, setCupon] = useState(null)
  const [cargado, setCargado] = useState(false)

  // Sesión con Google
  const [usuario, setUsuario] = useState(null)
  const [sesionLista, setSesionLista] = useState(false)

  // Estado compartido del panel del carrito (abierto / cerrado)
  const [panelAbierto, setPanelAbierto] = useState(false)

  function abrirPanel() {
    setPanelAbierto(true)
  }

  function cerrarPanel() {
    setPanelAbierto(false)
  }

  function alternarPanel() {
    setPanelAbierto((abierto) => !abierto)
  }

  // Al montar en el navegador, recuperar lo guardado
  useEffect(() => {
    try {
      const itemsGuardados = localStorage.getItem(CLAVE_ITEMS)
      const zonaGuardada = localStorage.getItem(CLAVE_ZONA)
      const cuponGuardado = localStorage.getItem(CLAVE_CUPON)

      if (itemsGuardados) setItems(JSON.parse(itemsGuardados))
      if (zonaGuardada) setZonaEnvio(JSON.parse(zonaGuardada))
      if (cuponGuardado) {
        const cuponParseado = JSON.parse(cuponGuardado)
        // No restaurar códigos antiguos: BIENVENIDA5 y CUPON10 se reemplazaron.
        if (!['BIENVENIDA5', 'CUPON10'].includes(String(cuponParseado?.codigo || '').toUpperCase())) {
          setCupon(cuponParseado)
        } else {
          localStorage.removeItem(CLAVE_CUPON)
        }
      }
    } catch (err) {
      console.error('Error al leer el carrito guardado:', err)
    } finally {
      setCargado(true)
    }
  }, [])

  // Escuchar la sesión de Google (al entrar, al volver de Google y al salir)
  useEffect(() => {
    let activo = true

    supabase.auth.getSession().then(({ data }) => {
      if (!activo) return
      setUsuario(data.session?.user ?? null)
      setSesionLista(true)
    })

    const { data: suscripcion } = supabase.auth.onAuthStateChange((_evento, sesion) => {
      setUsuario(sesion?.user ?? null)
      setSesionLista(true)
    })

    return () => {
      activo = false
      suscripcion.subscription.unsubscribe()
    }
  }, [])

  // Guardar items cada vez que cambian (una vez que ya se cargó lo inicial)
  useEffect(() => {
    if (!cargado) return
    try {
      localStorage.setItem(CLAVE_ITEMS, JSON.stringify(items))
    } catch (err) {
      console.error('Error al guardar el carrito:', err)
    }
  }, [items, cargado])

  // Guardar zona de envío cada vez que cambia
  useEffect(() => {
    if (!cargado) return
    try {
      if (zonaEnvio) {
        localStorage.setItem(CLAVE_ZONA, JSON.stringify(zonaEnvio))
      } else {
        localStorage.removeItem(CLAVE_ZONA)
      }
    } catch (err) {
      console.error('Error al guardar la zona de envío:', err)
    }
  }, [zonaEnvio, cargado])

  // Guardar el cupón cada vez que cambia
  useEffect(() => {
    if (!cargado) return
    try {
      if (cupon) {
        localStorage.setItem(CLAVE_CUPON, JSON.stringify(cupon))
      } else {
        localStorage.removeItem(CLAVE_CUPON)
      }
    } catch (err) {
      console.error('Error al guardar el cupón:', err)
    }
  }, [cupon, cargado])

  function agregarAlCarrito(producto, cantidad = 1) {
    setItems((prev) => {
      const existe = prev.find(
        (item) => item.id === producto.id && item.variante_id === producto.variante_id
      )
      if (existe) {
        return prev.map((item) =>
          item.id === producto.id && item.variante_id === producto.variante_id
            ? { ...item, cantidad: item.cantidad + cantidad }
            : item
        )
      }
      return [...prev, { ...producto, cantidad }]
    })
  }

  function quitarDelCarrito(id, varianteId) {
    setItems((prev) =>
      prev.filter((item) => !(item.id === id && item.variante_id === varianteId))
    )
  }

  function actualizarCantidad(id, varianteId, nuevaCantidad) {
    if (nuevaCantidad < 1) {
      quitarDelCarrito(id, varianteId)
      return
    }
    setItems((prev) =>
      prev.map((item) =>
        item.id === id && item.variante_id === varianteId
          ? { ...item, cantidad: nuevaCantidad }
          : item
      )
    )
  }

  function vaciarCarrito() {
    setItems([])
    setZonaEnvio(null)
    setCupon(null)
  }

  // ---------- Sesión con Google ----------

  async function iniciarSesionGoogle() {
    const volverA = window.location.href.split('#')[0]
    await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: { redirectTo: volverA },
    })
  }

  async function cerrarSesion() {
    await supabase.auth.signOut()
    setCupon(null)
  }

  // Cabeceras para las llamadas a la API: incluyen la sesión para que el servidor sepa quién es
  async function cabecerasConSesion() {
    const cabeceras = { 'Content-Type': 'application/json' }
    try {
      const { data } = await supabase.auth.getSession()
      const token = data.session?.access_token
      if (token) cabeceras.Authorization = `Bearer ${token}`
    } catch (err) {
      console.error('No se pudo leer la sesión:', err)
    }
    return cabeceras
  }

  // Comprueba con el servidor que el cupón exista y esté vigente, y lo deja cargado.
  // Para usar un cupón hay que haber ingresado con Google: si no, se recuerda el código
  // y se aplica solo apenas la persona ingresa.
  // El descuento definitivo se vuelve a calcular en el servidor al pagar.
  async function aplicarCupon(codigo) {
    try {
      const respuesta = await fetch('/api/validar-cupon', {
        method: 'POST',
        headers: await cabecerasConSesion(),
        body: JSON.stringify({ codigo }),
      })
      const datos = await respuesta.json()

      if (respuesta.status === 401 || datos.login_requerido) {
        try {
          localStorage.setItem(CLAVE_CUPON_PENDIENTE, normalizarCodigo(codigo))
        } catch {}
        return {
          ok: false,
          loginRequerido: true,
          error: datos.error || 'Para usar un cupón ingresá con Google.',
        }
      }

      if (!respuesta.ok || !datos.cupon) {
        return { ok: false, error: datos.error || 'No pudimos aplicar el cupón.' }
      }

      setCupon(datos.cupon)
      return { ok: true, cupon: datos.cupon }
    } catch (err) {
      return { ok: false, errorDeRed: true, error: 'No pudimos verificar el cupón. Probá de nuevo.' }
    }
  }

  function quitarCupon() {
    setCupon(null)
  }

  // Apenas la persona ingresa con Google:
  // 1) si había un cupón esperando (ej: venía de un QR), se aplica
  // 2) si no hay ningún cupón puesto, se prueba el de bienvenida (si ya lo usó, no pasa nada)
  useEffect(() => {
    if (!cargado || !sesionLista || !usuario) return

    let pendiente = null
    try {
      pendiente = localStorage.getItem(CLAVE_CUPON_PENDIENTE)
      if (pendiente) localStorage.removeItem(CLAVE_CUPON_PENDIENTE)
    } catch {}

    if (pendiente) {
      aplicarCupon(pendiente)
      return
    }

    if (!cupon) {
      const claveProbada = `${CLAVE_BIENVENIDA_PROBADA}_${usuario.id}`
      let yaProbada = false
      try {
        yaProbada = Boolean(sessionStorage.getItem(claveProbada))
      } catch {}
      if (!yaProbada) {
        aplicarCupon(CODIGO_BIENVENIDA).then((resultado) => {
          // Se marca como probada solo si el servidor contestó de verdad (lo aplicó o lo rechazó).
          // Si falló por red o por falta de sesión, se reintenta la próxima vez.
          if (resultado.ok || (!resultado.loginRequerido && !resultado.errorDeRed)) {
            try {
              sessionStorage.setItem(claveProbada, '1')
            } catch {}
          }
        })
      }
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [cargado, sesionLista, usuario])

  async function pagar(datosCliente, recaptchaToken) {
    const respuesta = await fetch('/api/crear-preferencia', {
      method: 'POST',
      headers: await cabecerasConSesion(),
      body: JSON.stringify({
        items,
        zonaEnvio,
        datosCliente,
        recaptchaToken,
        cuponCodigo: cupon?.codigo || null,
      }),
    })

    const datos = await respuesta.json()

    if (datos.init_point) {
      window.location.href = datos.init_point
    } else {
      const error = new Error(datos.error || 'No se pudo generar el link de pago')
      error.cuponInvalido = Boolean(datos.cupon_invalido)
      throw error
    }
  }

  const totalProductos = items.reduce((acc, item) => acc + item.precio * item.cantidad, 0)

  // Cupón: descuenta solo los productos y solo si se llega a la compra mínima
  const cuponAplica = Boolean(cupon) && totalProductos >= (Number(cupon.compra_minima) || 0)
  const descuentoCupon = cuponAplica ? calcularDescuentoCupon(cupon, totalProductos) : 0
  const totalProductosConCupon = totalProductos - descuentoCupon

  // Envío gratis desde cierto monto de productos, medido DESPUÉS del cupón (ver configEnvioGratis.js)
  const envioGratis = totalProductosConCupon >= UMBRAL_ENVIO_GRATIS
  const faltaParaEnvioGratis = Math.max(0, UMBRAL_ENVIO_GRATIS - totalProductosConCupon)
  const costoEnvio = calcularCostoEnvio(totalProductosConCupon, zonaEnvio)
  const total = totalProductosConCupon + costoEnvio
  const cantidadTotal = items.reduce((acc, item) => acc + item.cantidad, 0)

  return (
    <CarritoContext.Provider
      value={{
        items,
        agregarAlCarrito,
        quitarDelCarrito,
        actualizarCantidad,
        vaciarCarrito,
        total,
        totalProductos,
        totalProductosConCupon,
        costoEnvio,
        envioGratis,
        faltaParaEnvioGratis,
        cantidadTotal,
        pagar,
        zonaEnvio,
        setZonaEnvio,
        cupon,
        cuponAplica,
        descuentoCupon,
        aplicarCupon,
        quitarCupon,
        usuario,
        sesionLista,
        iniciarSesionGoogle,
        cerrarSesion,
        cabecerasConSesion,
        panelAbierto,
        abrirPanel,
        cerrarPanel,
        alternarPanel,
      }}
    >
      {children}
    </CarritoContext.Provider>
  )
}

// Archivo: src/lib/configCupones.js
// Funciones puras (sin base de datos): las usan tanto el navegador como el servidor.
//
// Reglas de los cupones:
// - Descuentan solo los productos, no el envío.
// - Se aplican ANTES del 5% de transferencia (el 5% se calcula sobre el precio ya con cupón).
// - El monto del envío gratis se mide DESPUÉS del cupón.

export function normalizarCodigo(texto) {
  return String(texto || '').trim().toUpperCase().replace(/\s+/g, '')
}

// Cuánto descuenta el cupón sobre el subtotal de productos. Nunca más que el subtotal.
export function calcularDescuentoCupon(cupon, subtotal) {
  if (!cupon || !(subtotal > 0)) return 0
  const valor = Number(cupon.valor) || 0
  const bruto = cupon.tipo === 'porcentaje' ? Math.round((subtotal * valor) / 100) : Math.round(valor)
  return Math.max(0, Math.min(bruto, subtotal))
}

export function etiquetaDescuento(cupon) {
  if (cupon.tipo === 'porcentaje') return `−${Number(cupon.valor)}%`
  return `−$${Number(cupon.valor).toLocaleString('es-AR')}`
}

// Texto de la línea que queda en el pedido, el admin y los emails
export function nombreLineaCupon(cupon) {
  return `Cupón ${cupon.codigo} (${etiquetaDescuento(cupon)})`
}

// Envío gratis a partir de cierto monto.
// Archivo: src/lib/configEnvioGratis.js
//
// El monto se compara contra el total de los PRODUCTOS (antes del descuento por transferencia
// y sin contar el envío). Para cambiar el monto, editá UMBRAL_ENVIO_GRATIS y hacé git push.

export const UMBRAL_ENVIO_GRATIS = 150000

// Cuando el envío es gratis, el cliente NO elige empresa: el sistema la elige solo.
// Orden de preferencia (siempre a domicilio). Se usa la primera que tenga tarifa para su código postal.
export const EMPRESAS_ENVIO_GRATIS = ['urbano', 'oca']

// Dentro de cada empresa se prefiere este servicio (OCA lo llama "Sucursal a puerta").
// Si no aparece, se usa el más barato de los que llegan a la puerta del cliente.
export const SERVICIOS_PREFERIDOS = ['sucursal a puerta']

// Si ninguna de las anteriores llega a ese código postal, se usa la a domicilio más barata de
// cualquier otra empresa, para no perder la venta. Poné false si preferís que no se pueda comprar.
export const USAR_OTRAS_EMPRESAS_SI_FALTAN = true

// Lo que se guarda como "zona de envío" cuando el envío es gratis (el cliente no ve la empresa)
export const ID_ENVIO_GRATIS_AUTO = 'envio-gratis-auto'
export const NOMBRE_ENVIO_GRATIS = 'Envío gratis a domicilio'

// Devuelve el costo de envío que realmente se cobra
export function calcularCostoEnvio(subtotalProductos, zonaEnvio) {
  if (!zonaEnvio) return 0
  if (subtotalProductos >= UMBRAL_ENVIO_GRATIS) return 0
  return Number(zonaEnvio.costo) || 0
}

function normalizar(texto) {
  return String(texto || '')
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
}

// Una opción es "a domicilio" si el cliente la recibe en su puerta.
// OCA: "Sucursal a puerta" y "Puerta a puerta" llegan al domicilio; "Sucursal a sucursal" y
// "Puerta a sucursal" no (el destino es una sucursal).
function esADomicilio(opcion) {
  const texto = `${normalizar(opcion.service)} ${normalizar(opcion.serviceCode)}`
  if (/\ba\s+(sucursal|agencia|punto|locker)/.test(texto)) return false
  if (/puerta|domicilio|home|door/.test(texto)) return true
  if (/sucursal|agencia|punto|retiro|pickup|locker/.test(texto)) return false
  if (opcion.branches?.length > 0) return false
  return true
}

function esServicioPreferido(opcion) {
  const texto = `${normalizar(opcion.service)} ${normalizar(opcion.serviceCode)}`
  return SERVICIOS_PREFERIDOS.some((s) => texto.includes(s))
}

// Primero el servicio preferido y, entre iguales, el más barato
function ordenar(opciones) {
  return [...opciones].sort(
    (a, b) =>
      Number(esServicioPreferido(b)) - Number(esServicioPreferido(a)) || precioDe(a) - precioDe(b)
  )
}

function precioDe(opcion) {
  return Number(opcion.price) || 0
}

// De todas las tarifas que devuelve la cotización, elige la que se usa cuando el envío es gratis
export function elegirOpcionEnvioGratis(opciones) {
  const aDomicilio = (opciones || []).filter(esADomicilio)

  for (const empresa of EMPRESAS_ENVIO_GRATIS) {
    const candidatas = aDomicilio.filter((o) =>
      `${normalizar(o.carrierSlug)} ${normalizar(o.carrier)}`.includes(empresa)
    )
    if (candidatas.length > 0) {
      return ordenar(candidatas)[0]
    }
  }

  if (USAR_OTRAS_EMPRESAS_SI_FALTAN && aDomicilio.length > 0) {
    return ordenar(aDomicilio)[0]
  }

  return null
}

// DESTINO: src/lib/cupones.js  (reemplaza el archivo actual)
// SOLO para el servidor (rutas api): valida cupones, anota sus usos y reconoce
// a la persona que ingresó con Google.

import { createClient } from '@supabase/supabase-js'
import { normalizarCodigo, calcularDescuentoCupon } from '@/lib/configCupones'

const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY
)

// Mensaje único para todos los casos, así nadie puede averiguar qué códigos existen
const MENSAJE_NO_VALIDO = 'El cupón no es válido o ya no está disponible.'
export const MENSAJE_LOGIN = 'Para usar un cupón ingresá con Google.'
export const MENSAJE_YA_USADO = 'Ya usaste este cupón con tu cuenta.'

// Lee la sesión de Google que el navegador manda en la cabecera Authorization.
// Devuelve el usuario, o null si no hay sesión válida.
export async function usuarioDesdeRequest(request) {
  const cabecera = request.headers.get('authorization') || ''
  const token = cabecera.startsWith('Bearer ') ? cabecera.slice(7).trim() : ''
  if (!token) return null

  const { data, error } = await supabaseAdmin.auth.getUser(token)
  if (error || !data?.user) return null
  return data.user
}

// ¿Esta cuenta ya usó este cupón?
async function cuentaYaUsoCupon(userId, codigo) {
  const { data, error } = await supabaseAdmin
    .from('cupones_usos')
    .select('id')
    .eq('user_id', userId)
    .eq('codigo', codigo)
    .maybeSingle()

  if (error) {
    console.error('Error al revisar los usos del cupón:', error)
    return false
  }
  return Boolean(data)
}

// Comprueba que el cupón exista y esté vigente (sin mirar el monto de la compra).
// Si se pasa userId, también comprueba que esa cuenta no lo haya usado antes.
export async function buscarCuponValido(codigoCrudo, userId = null) {
  const codigo = normalizarCodigo(codigoCrudo)
  if (!codigo) return { ok: false, error: 'Ingresá un código de cupón.' }

  const { data: cupon, error } = await supabaseAdmin
    .from('cupones')
    .select('*')
    .eq('codigo', codigo)
    .maybeSingle()

  if (error) {
    console.error('Error al buscar el cupón:', error)
    return { ok: false, error: 'No pudimos verificar el cupón. Probá de nuevo.' }
  }

  if (!cupon || !cupon.activo) return { ok: false, error: MENSAJE_NO_VALIDO }

  const ahora = new Date()
  if (cupon.fecha_inicio && new Date(cupon.fecha_inicio) > ahora) return { ok: false, error: MENSAJE_NO_VALIDO }
  if (cupon.fecha_fin && new Date(cupon.fecha_fin) < ahora) return { ok: false, error: MENSAJE_NO_VALIDO }
  if (cupon.max_usos != null && cupon.usos >= cupon.max_usos) return { ok: false, error: MENSAJE_NO_VALIDO }

  if (userId && (await cuentaYaUsoCupon(userId, cupon.codigo))) {
    return { ok: false, error: MENSAJE_YA_USADO }
  }

  return { ok: true, cupon }
}

// Valida el cupón contra el subtotal real del pedido y calcula el descuento
export async function validarCuponParaPedido(codigo, subtotal, userId = null) {
  const resultado = await buscarCuponValido(codigo, userId)
  if (!resultado.ok) return resultado

  const minimo = Number(resultado.cupon.compra_minima) || 0
  if (subtotal < minimo) {
    return {
      ok: false,
      error: `Este cupón requiere una compra mínima de $${minimo.toLocaleString('es-AR')} en productos.`,
    }
  }

  const descuento = calcularDescuentoCupon(resultado.cupon, subtotal)
  if (descuento <= 0) return { ok: false, error: MENSAJE_NO_VALIDO }

  return { ok: true, cupon: resultado.cupon, descuento }
}

// Suma un uso de forma atómica. Devuelve false si el cupón se agotó justo en ese momento.
export async function usarCupon(codigo) {
  const { data, error } = await supabaseAdmin.rpc('usar_cupon', { p_codigo: codigo })
  if (error) {
    console.error('Error al registrar el uso del cupón:', error)
    return false
  }
  return data === true
}

// Devuelve el uso si el pedido no se pudo completar
export async function devolverUsoCupon(codigo) {
  const { error } = await supabaseAdmin.rpc('devolver_uso_cupon', { p_codigo: codigo })
  if (error) console.error('Error al devolver el uso del cupón:', error)
}

// Anota que esta cuenta usó el cupón. Devuelve 'ok', 'ya_usado' o 'error'.
// El índice único de la base impide que una misma cuenta lo use dos veces,
// aunque dos pedidos lleguen al mismo tiempo.
export async function reservarUsoCuenta(userId, codigo, pedidoRef = null) {
  const { error } = await supabaseAdmin.from('cupones_usos').insert({
    user_id: userId,
    codigo,
    pedido_id: pedidoRef != null ? String(pedidoRef) : null,
  })

  if (!error) return 'ok'
  if (error.code === '23505') return 'ya_usado'
  console.error('Error al anotar el uso del cupón por cuenta:', error)
  return 'error'
}

// Deshace la anotación si el pedido no se pudo completar
export async function liberarUsoCuenta(userId, codigo) {
  const { error } = await supabaseAdmin
    .from('cupones_usos')
    .delete()
    .eq('user_id', userId)
    .eq('codigo', codigo)
  if (error) console.error('Error al liberar el uso del cupón por cuenta:', error)
}

    .eq('codigo', codigo)
  if (error) console.error('Error al vincular el uso del cupón con el pedido:', error)
}

// Mercado Pago no acepta líneas con precio negativo: el descuento se reparte entre los
// precios de los productos, proporcionalmente, y la suma queda igual al total con cupón.
export function repartirDescuentoEnItems(items, descuento) {
  if (!descuento || descuento <= 0) return items

  const subtotal = items.reduce((acc, i) => acc + i.unit_price * i.quantity, 0)
  if (subtotal <= 0) return items

  let restante = descuento
  return items.map((item, idx) => {
    const totalLinea = item.unit_price * item.quantity
    const esUltima = idx === items.length - 1
    let descLinea = esUltima ? restante : Math.round((descuento * totalLinea) / subtotal)
    descLinea = Math.max(0, Math.min(descLinea, totalLinea, restante))
    restante -= descLinea
    const nuevoPrecio = Math.round(((totalLinea - descLinea) / item.quantity) * 100) / 100
    return { ...item, unit_price: nuevoPrecio }
  })
}

function formatearItems(items) {
  return items
    .map((item) => {
      const foto = item.imagen_url
        ? `<td style="width: 64px; padding-right: 12px; vertical-align: top;">
             <table role="presentation" width="64" height="64" style="width: 64px; height: 64px; background-color: #FAF6F0; border: 1px solid #E4DCCF; border-radius: 8px;">
               <tr>
                 <td align="center" valign="middle" style="text-align: center; vertical-align: middle;">
                   <img src="${item.imagen_url}" alt="${item.nombre}" width="56" style="max-width: 56px; max-height: 56px; display: block; margin: 0 auto;" />
                 </td>
               </tr>
             </table>
           </td>`
        : ''

      // Una línea con precio negativo es un descuento (ej: transferencia)
      const subtotal = item.precio * item.cantidad
      const importe = subtotal < 0 ? `-$${Math.abs(subtotal)}` : `$${subtotal}`
      const detalle = subtotal < 0 ? importe : `Cantidad: ${item.cantidad} — ${importe}`

      return `
        <table role="presentation" width="100%" style="margin-bottom: 10px;">
          <tr>
            ${foto}
            <td style="vertical-align: top; font-size: 14px; color: #2E2A26;">
              <div>${item.nombre}</div>
              <div style="color: #8A8378;">${detalle}</div>
            </td>
          </tr>
        </table>
      `
    })
    .join('')
}

// Bloque de entrega para los emails al cliente: tiempo de confección y, si corresponde, aviso de retiro
function bloqueEntrega(pedido) {
  const esRetiro = pedido.zona_envio === 'Retiro en persona'
  return `
          <p style="color: #2E2A26;"><strong>Tiempo de confección estimado:</strong> ${textoConfeccion()}.</p>
          ${esRetiro ? '<p style="color: #2E2A26;"><strong>Retiro en persona:</strong> te avisamos por WhatsApp o por email cuando tu pedido esté listo, y coordinamos el día y el horario para retirarlo.</p>' : ''}`
}

   function lineaEnvio(pedido) {
     if (!pedido.zona_envio) return ''
     const esRetiro = pedido.zona_envio === 'Retiro en persona'
     const texto =
       Number(pedido.costo_envio) === 0 && !esRetiro
         ? 'Costo de envío: ¡Gratis!'
         : `Envío (${pedido.zona_envio}): $${pedido.costo_envio}`
     return `<p style="margin: 10px 0 0; color: #2E2A26;">${texto}</p>`
   }

function formatearDireccion(direccion) {
  if (!direccion) return ''
  const partes = [
    `${direccion.calle} ${direccion.numero}`,
    direccion.pisoDepto ? direccion.pisoDepto : null,
    direccion.ciudad,
    direccion.codigoPostal ? `CP ${direccion.codigoPostal}` : null,
  ].filter(Boolean)
  return partes.join(', ')
}
