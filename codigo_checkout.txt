===== src\app\admin\BotonConfirmarTransferencia.js =====
'use client'

import { useState } from 'react'

export default function BotonConfirmarTransferencia({ pedido }) {
  const [cargando, setCargando] = useState(false)
  const [resultado, setResultado] = useState(null)

  const esTransferencia = pedido.metodo_pago === 'transferencia'
  const pendiente = pedido.estado === 'pending'
  // Pagado pero la factura falló: el mismo botón la reintenta
  const faltaFactura = pedido.estado === 'approved' && !pedido.factura_cae

  if (!esTransferencia || (!pendiente && !faltaFactura)) return null

  async function confirmar() {
    const aviso = pendiente
      ? `¿Ya viste $${pedido.total} acreditados en tu cuenta por el pedido #${pedido.id}?\n\nAl confirmar se marca como pagado, se emite la factura real en AFIP y se le avisa al cliente por email.`
      : '¿Reintentar la factura de este pedido? Se emite una factura real en AFIP.'
    if (!confirm(aviso)) return

    setCargando(true)
    setResultado(null)

    try {
      const res = await fetch('/api/admin/pedidos-confirmar-transferencia', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ pedidoId: pedido.id }),
      })

      const data = await res.json()

      if (!res.ok || !data.success) {
        setResultado({ tipo: 'error', mensaje: data.error || 'Error al confirmar el pago.' })
      } else if (!data.facturaOk) {
        setResultado({
          tipo: 'error',
          mensaje: 'Pago confirmado, pero la factura falló. Recargá la página y tocá "Reintentar factura".',
        })
      } else {
        setResultado({ tipo: 'ok', mensaje: 'Pago confirmado y factura emitida. Recargá la página.' })
      }
    } catch (err) {
      setResultado({ tipo: 'error', mensaje: 'Error de conexión.' })
    } finally {
      setCargando(false)
    }
  }

  return (
    <div style={{ marginTop: '10px' }}>
      <button
        onClick={confirmar}
        disabled={cargando}
        style={{
          padding: '8px 14px',
          fontSize: '13px',
          fontWeight: '600',
          backgroundColor: cargando ? '#ccc' : '#2F6B63',
          color: 'white',
          border: 'none',
          borderRadius: '6px',
          cursor: cargando ? 'not-allowed' : 'pointer',
        }}
      >
        {cargando ? 'Confirmando...' : pendiente ? '🏦 Confirmar pago' : '🧾 Reintentar factura'}
      </button>
      {resultado && (
        <p style={{ marginTop: '6px', fontSize: '13px', color: resultado.tipo === 'ok' ? '#15803d' : '#dc2626' }}>
          {resultado.mensaje}
        </p>
      )}
    </div>
  )
}

const CODIGO_VALIDO = /^[A-Z0-9_-]{3,30}$/

function volver(mensaje, tipo = 'error') {
  redirect(`/admin/cupones?${tipo}=${encodeURIComponent(mensaje)}`)
}

// Estas acciones corren en el servidor y están dentro de /admin/cupones,
// así que quedan protegidas igual que el resto del panel.
async function crearCupon(formData) {
  'use server'

  const codigo = normalizarCodigo(formData.get('codigo'))
  const tipo = formData.get('tipo') === 'monto' ? 'monto' : 'porcentaje'
  const valor = Number(formData.get('valor'))
  const maxUsosTexto = String(formData.get('max_usos') || '').trim()
  const compraMinima = Number(formData.get('compra_minima')) || 0
  const desde = String(formData.get('fecha_inicio') || '').trim()
  const hasta = String(formData.get('fecha_fin') || '').trim()

  if (!CODIGO_VALIDO.test(codigo)) {
    volver('El código debe tener entre 3 y 30 caracteres: letras, números, guión o guión bajo.')
  }
  if (!(valor > 0)) volver('Ingresá un valor mayor a 0.')
  if (tipo === 'porcentaje' && valor > 100) volver('El porcentaje no puede superar 100.')

  const maxUsos = maxUsosTexto ? Math.floor(Number(maxUsosTexto)) : null
  if (maxUsosTexto && !(maxUsos > 0)) volver('La cantidad máxima de usos tiene que ser un número mayor a 0.')
  if (compraMinima < 0) volver('La compra mínima no puede ser negativa.')
  if (desde && hasta && hasta < desde) volver('La fecha de fin no puede ser anterior a la de inicio.')

  const { error } = await supabaseAdmin.from('cupones').insert({
    codigo,
    tipo,
    valor,
    max_usos: maxUsos,
    compra_minima: compraMinima,
    // Horario de Argentina: empieza a las 00:00 del día de inicio y termina a las 23:59 del día de fin
    fecha_inicio: desde ? `${desde}T00:00:00-03:00` : null,
    fecha_fin: hasta ? `${hasta}T23:59:59-03:00` : null,
  })

  if (error) {
    console.error('Error al crear el cupón:', error)
    volver(error.code === '23505' ? 'Ya existe un cupón con ese código.' : 'No se pudo crear el cupón.')
  }

  revalidatePath('/admin/cupones')
  volver('Cupón creado.', 'ok')
}

async function alternarCupon(formData) {
  'use server'
  const id = Number(formData.get('id'))
  const estabaActivo = formData.get('activo') === 'true'
  await supabaseAdmin.from('cupones').update({ activo: !estabaActivo }).eq('id', id)
  revalidatePath('/admin/cupones')
}

// Solo se pueden eliminar cupones apagados, para no borrar uno activo por error
async function eliminarCupon(formData) {
  'use server'
  const id = Number(formData.get('id'))
  await supabaseAdmin.from('cupones').delete().eq('id', id).eq('activo', false)
  revalidatePath('/admin/cupones')
}

function estadoDe(cupon, ahora) {
  if (!cupon.activo) return { label: 'Apagado', fondo: '#f3f4f6', texto: '#6b7280' }
  if (cupon.fecha_fin && new Date(cupon.fecha_fin) < ahora) return { label: 'Vencido', fondo: '#fee2e2', texto: '#b91c1c' }
  if (cupon.max_usos != null && cupon.usos >= cupon.max_usos) return { label: 'Agotado', fondo: '#fee2e2', texto: '#b91c1c' }
  if (cupon.fecha_inicio && new Date(cupon.fecha_inicio) > ahora) return { label: 'Aún no empieza', fondo: '#fef9c3', texto: '#a16207' }
  return { label: 'Activo', fondo: '#dcfce7', texto: '#15803d' }
}

function fechaCorta(valor) {
  return new Date(valor).toLocaleDateString('es-AR', { timeZone: 'America/Argentina/Buenos_Aires' })
}

===== src\app\admin\cupones\QrCupon.js =====
'use client'

import { useEffect, useState } from 'react'
import QRCode from 'qrcode'

// Genera el QR de un cupón en el navegador y permite descargarlo para imprimir
export default function QrCupon({ url, codigo }) {
  const [src, setSrc] = useState(null)

  useEffect(() => {
    QRCode.toDataURL(url, { width: 600, margin: 2, errorCorrectionLevel: 'M' })
      .then(setSrc)
      .catch(() => setSrc(null))
  }, [url])

  if (!src) return null

  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: '12px', marginTop: '12px' }}>
      <img src={src} alt={`QR del cupón ${codigo}`} width={110} height={110} style={{ border: '1px solid #e5e5e5', borderRadius: '6px' }} />
      <div style={{ fontSize: '13px', color: '#666' }}>
        <p style={{ margin: '0 0 6px', wordBreak: 'break-all' }}>{url}</p>
        <a
          href={src}
          download={`qr-cupon-${codigo}.png`}
          style={{ color: '#2F6B63', fontWeight: 600 }}
        >
          ⬇️ Descargar QR
        </a>
      </div>
    </div>
  )
}

export async function POST(request) {
  // Si el pedido no se llega a guardar, se devuelven los usos que se habían reservado
  let cuponReservado = null
  let usoCuentaReservado = null // { userId, codigo }

  async function deshacerReservas() {
    if (cuponReservado) {
      await devolverUsoCupon(cuponReservado)
      cuponReservado = null
    }
    if (usoCuentaReservado) {
      await liberarUsoCuenta(usoCuentaReservado.userId, usoCuentaReservado.codigo)
      usoCuentaReservado = null
    }
  }

  try {
    const { items, zonaEnvio, datosCliente, recaptchaToken, cuponCodigo } = await request.json()

    const recaptchaValido = await verificarRecaptcha(recaptchaToken)
    if (!recaptchaValido) {
      return NextResponse.json(
        { error: 'No pudimos verificar que sos una persona real. Recargá la página e intentá de nuevo.' },
        { status: 400 }
      )
    }

    if (!Array.isArray(items) || items.length === 0) {
      return NextResponse.json({ error: 'Tu carrito está vacío' }, { status: 400 })
    }
    if (!zonaEnvio) {
      return NextResponse.json({ error: 'Elegí una zona de envío' }, { status: 400 })
    }
    if (!datosCliente?.nombre || !datosCliente?.email) {
      return NextResponse.json({ error: 'Faltan tus datos de contacto' }, { status: 400 })
    }

    // Mismo criterio que crear-preferencia: se toman nombre y precio del carrito.
    // Se limpian cantidad y precio para que no entren valores raros.
    const itemsPedido = items.map((item) => {
      const cantidad = Math.max(1, parseInt(item.cantidad, 10) || 1)
      const precio = Math.max(0, Number(item.precio) || 0)
      const detalles = [item.color, item.talle ? `Talle ${item.talle}` : null].filter(Boolean).join(' - ')

      return {
        nombre: detalles ? `${item.nombre} (${detalles})` : item.nombre,
        cantidad,
        precio,
        imagen_url: item.imagen_url || null,
      }
    })

    const subtotal = itemsPedido.reduce((acc, item) => acc + item.precio * item.cantidad, 0)

    // El cupón y el descuento por transferencia SE SUMAN: primero se descuenta el cupón
    // y el % de transferencia se calcula sobre lo que queda (5% + 5% = 9,75%).
    let cupon = null
    let descuentoCupon = 0

    if (cuponCodigo) {
      // Para usar un cupón hay que haber ingresado con Google
      const usuario = await usuarioDesdeRequest(request)
      if (!usuario) {
        return NextResponse.json(
          { error: MENSAJE_LOGIN, cupon_invalido: true, login_requerido: true },
          { status: 401 }
        )
      }

      // El cupón se valida y se calcula acá, nunca se confía en el descuento que manda el navegador
      const validacion = await validarCuponParaPedido(cuponCodigo, subtotal, usuario.id)
      if (!validacion.ok) {
        return NextResponse.json({ error: validacion.error, cupon_invalido: true }, { status: 400 })
      }

      const reservado = await usarCupon(validacion.cupon.codigo)
      if (!reservado) {
        return NextResponse.json(
          { error: 'El cupón no es válido o ya no está disponible.', cupon_invalido: true },
          { status: 400 }
        )
      }
      cuponReservado = validacion.cupon.codigo

      const uso = await reservarUsoCuenta(usuario.id, validacion.cupon.codigo)
      if (uso !== 'ok') {
        await deshacerReservas()
        return NextResponse.json(
          {
            error: uso === 'ya_usado' ? MENSAJE_YA_USADO : 'No pudimos verificar el cupón. Probá de nuevo.',
            cupon_invalido: true,
          },
          { status: 400 }
        )
      }
      usoCuentaReservado = { userId: usuario.id, codigo: validacion.cupon.codigo }

      cupon = validacion.cupon
      descuentoCupon = validacion.descuento
    }

    const subtotalConCupon = subtotal - descuentoCupon
    const descuento = Math.round(subtotalConCupon * DESCUENTO_TRANSFERENCIA)
    const costoEnvio = calcularCostoEnvio(subtotalConCupon, zonaEnvio)
    const total = subtotalConCupon - descuento + costoEnvio

    // Los descuentos quedan como líneas con precio negativo, así todo suma igual
    // en el admin, en los emails y en la factura
    if (descuentoCupon > 0) {
      itemsPedido.push({
        nombre: nombreLineaCupon(cupon),
        cantidad: 1,
        precio: -descuentoCupon,
        imagen_url: null,
      })
    }

    if (descuento > 0) {
      itemsPedido.push({
        nombre: `Descuento por transferencia (${Math.round(DESCUENTO_TRANSFERENCIA * 100)}%)`,
        cantidad: 1,
        precio: -descuento,
        imagen_url: null,
      })
    }

    const direccion = datosCliente.calle
      ? {
          calle: datosCliente.calle,
          numero: datosCliente.numero,
          pisoDepto: datosCliente.pisoDepto || null,
          ciudad: datosCliente.ciudad,
          codigoPostal: datosCliente.codigoPostal || null,
        }
      : null

    // La tabla identifica cada pedido por esta columna: se usa una referencia propia que arranca con TR-
    const referencia =
      'TR-' + Date.now().toString(36).toUpperCase() + Math.random().toString(36).slice(2, 6).toUpperCase()

    const { data: pedido, error: errorInsert } = await supabaseAdmin
      .from('pedidos')
      .insert({
        mercadopago_payment_id: referencia,
        estado: 'pending',
        metodo_pago: 'transferencia',
        descuento,
        cupon_codigo: cupon ? cupon.codigo : null,
        descuento_cupon: descuentoCupon,
        total,
        items: itemsPedido,
        comprador_email: datosCliente.email || null,
        cliente_nombre: datosCliente.nombre || null,
        cliente_dni: datosCliente.dni || null,
        cliente_telefono: datosCliente.telefono || null,
        direccion,
        zona_envio: zonaEnvio.nombre || null,
        costo_envio: costoEnvio,
        envio_carrier: zonaEnvio.envioCarrier || zonaEnvio.envio_carrier || null,
        envio_service_code: zonaEnvio.envioServiceCode || zonaEnvio.envio_service_code || null,
        envio_branch_code: zonaEnvio.envioBranchCode || zonaEnvio.envio_branch_code || null,
        notas: datosCliente.notas || null,
      })
      .select('id')
      .single()

    if (errorInsert) {
      console.error('Error al guardar el pedido por transferencia:', errorInsert)
      await deshacerReservas()
      return NextResponse.json({ error: 'No se pudo guardar el pedido. Probá de nuevo.' }, { status: 500 })
    }

    // El pedido ya está guardado: los usos del cupón quedan firmes aunque falle algo después (por ejemplo un email)
    if (usoCuentaReservado) {
      await vincularUsoConPedido(usoCuentaReservado.userId, usoCuentaReservado.codigo, pedido.id)
    }
    cuponReservado = null
    usoCuentaReservado = null

    const pedidoParaEmail = {
      id: pedido.id,
      cliente_nombre: datosCliente.nombre || null,
      comprador_email: datosCliente.email || null,
      cliente_telefono: datosCliente.telefono || null,
      cliente_dni: datosCliente.dni || null,
      items: itemsPedido,
      total,
      zona_envio: zonaEnvio.nombre || null,
      costo_envio: costoEnvio,
      direccion,
      notas: datosCliente.notas || null,
    }

    await enviarEmailInstruccionesTransferencia(pedidoParaEmail)
    await enviarEmailNotificacionTransferencia(pedidoParaEmail)

    return NextResponse.json({ pedido_id: pedido.id, total, descuento })
  } catch (error) {
    console.error('Error al crear pedido por transferencia:', error)
    await deshacerReservas()
    return NextResponse.json({ error: 'Error al crear el pedido' }, { status: 500 })
  }
}

export async function POST(request) {
  // Si el pago no se llega a crear, se devuelve el uso del cupón que se había reservado
  let cuponReservado = null

  try {
    const { items, zonaEnvio, datosCliente, recaptchaToken, cuponCodigo } = await request.json()

    const recaptchaValido = await verificarRecaptcha(recaptchaToken)
    if (!recaptchaValido) {
      return NextResponse.json(
        { error: 'No pudimos verificar que sos una persona real. Recargá la página e intentá de nuevo.' },
        { status: 400 }
      )
    }

    const subtotalProductos = items.reduce((acc, item) => acc + item.precio * item.cantidad, 0)

    // Cupón: se valida y se calcula acá, nunca se confía en el descuento que manda el navegador.
    // Para usar un cupón hay que haber ingresado con Google.
    let cupon = null
    let descuentoCupon = 0
    let usuario = null

    if (cuponCodigo) {
      usuario = await usuarioDesdeRequest(request)
      if (!usuario) {
        return NextResponse.json(
          { error: MENSAJE_LOGIN, cupon_invalido: true, login_requerido: true },
          { status: 401 }
        )
      }

      const validacion = await validarCuponParaPedido(cuponCodigo, subtotalProductos, usuario.id)
      if (!validacion.ok) {
        return NextResponse.json({ error: validacion.error, cupon_invalido: true }, { status: 400 })
      }
      if (validacion.descuento >= subtotalProductos) {
        return NextResponse.json(
          { error: 'Este cupón cubre todo el pedido y no se puede usar con Mercado Pago. Escribinos y lo resolvemos.', cupon_invalido: true },
          { status: 400 }
        )
      }

      const reservado = await usarCupon(validacion.cupon.codigo)
      if (!reservado) {
        return NextResponse.json(
          { error: 'El cupón no es válido o ya no está disponible.', cupon_invalido: true },
          { status: 400 }
        )
      }

      cuponReservado = validacion.cupon.codigo
      cupon = validacion.cupon
      descuentoCupon = validacion.descuento
    }

    const preference = new Preference(client)

    const itemsParaMP = items.map((item) => {
      const detalles = [item.color, item.talle ? `Talle ${item.talle}` : null]
        .filter(Boolean)
        .join(' - ')

      return {
        id: String(item.id),
        title: detalles ? `${item.nombre} (${detalles})` : item.nombre,
        quantity: item.cantidad,
        unit_price: item.precio,
        currency_id: 'ARS',
      }
    })

    // Mercado Pago no acepta líneas con precio negativo: el descuento del cupón se reparte
    // entre los precios de los productos (el total queda correcto)
    const itemsConCupon = repartirDescuentoEnItems(itemsParaMP, descuentoCupon)

    // Envío gratis desde cierto monto (medido después del cupón): el costo se recalcula acá,
    // no se confía en el del navegador
    const subtotalConCupon = subtotalProductos - descuentoCupon
    const costoEnvio = calcularCostoEnvio(subtotalConCupon, zonaEnvio)

    if (zonaEnvio && costoEnvio > 0) {
      itemsConCupon.push({
        id: 'envio',
        title: `Envío - ${zonaEnvio.nombre}`,
        quantity: 1,
        unit_price: costoEnvio,
        currency_id: 'ARS',
      })
    }

    // Guardamos también los items originales (con imagen_url) en metadata,
    // porque Mercado Pago no permite campos custom dentro de "items".
    const itemsOriginales = items.map((item) => ({
      nombre: item.nombre,
      color: item.color || null,
      talle: item.talle || null,
      imagen_url: item.imagen_url || null,
    }))

    const resultado = await preference.create({
      body: {
        items: itemsConCupon,
        payer: {
          name: datosCliente?.nombre,
          email: datosCliente?.email,
        },
        metadata: {
          datos_cliente: datosCliente,
          zona_envio: zonaEnvio ? { ...zonaEnvio, costo: costoEnvio } : null,
          items_originales: itemsOriginales,
          cupon: cupon ? { codigo: cupon.codigo, descuento: descuentoCupon } : null,
          // Quién usó el cupón: al aprobarse el pago, el webhook lo anota en esa cuenta
          usuario_id: usuario ? usuario.id : null,
        },
        back_urls: {
          success: 'https://www.dimedetiambos.com.ar/pago-exitoso',
          failure: 'https://www.dimedetiambos.com.ar/pago-fallido',
          pending: 'https://www.dimedetiambos.com.ar/pago-pendiente',
        },
        auto_return: 'approved',
        notification_url: 'https://www.dimedetiambos.com.ar/api/webhook-mercadopago',
      },
    })

    return NextResponse.json({ id: resultado.id, init_point: resultado.init_point })
  } catch (error) {
    console.error('Error al crear preferencia:', error)
    if (cuponReservado) await devolverUsoCupon(cuponReservado)
    return NextResponse.json({ error: 'Error al crear la preferencia de pago' }, { status: 500 })
  }

export async function GET(request) {

  const mode = searchParams.get("hub.mode");

export async function POST(request) {
  const body = await request.json();

  // Estados de entrega de WhatsApp (sent, delivered, read, failed)
  const statuses = body?.entry?.[0]?.changes?.[0]?.value?.statuses;
  if (statuses) console.log("STATUS WhatsApp:", JSON.stringify(statuses));

  try {
    const entry = body.entry?.[0];
    const changes = entry?.changes?.[0];
    const message = changes?.value?.messages?.[0];

    if (!message) {
      return new Response("OK", { status: 200 });
    }

    // Meta reenvía el aviso si tardamos en responder: si este mensaje ya lo guardamos, no lo repetimos
    if (message.id) {
      const { data: yaGuardado } = await supabase
        .from("whatsapp_conversaciones")
        .select("id")
        .eq("wa_id", message.id)
        .limit(1)
        .maybeSingle();

      if (yaGuardado) {
        return new Response("OK", { status: 200 });
      }
    }

    const from = message.from;
    const paraEnviar = from.startsWith("549") ? "54" + from.slice(3) : from;
    const texto = message.text?.body;

    if (texto) {
      const hiloId = await obtenerOCrearHilo(from);
      await asegurarCliente(from);
      await guardarMensaje(from, hiloId, "user", texto, { wa_id: message.id });
      await avisarNuevoMensaje(from, texto);

      const pausado = await estaBotPausado(from);

      if (!pausado && BOT_ACTIVO) {
        const historial = await obtenerHistorialDelHilo(hiloId);
        const respuesta = await preguntarleAClaude(historial);

        await guardarMensaje(from, hiloId, "model", respuesta);
        await mandarMensajeWhatsApp(paraEnviar, respuesta);
      }
    } else if (ETIQUETAS_MEDIA[message.type]) {
      await procesarArchivoRecibido(message, from, paraEnviar);
    }

    return new Response("OK", { status: 200 });
  } catch (error) {
    console.error("Error en webhook de WhatsApp:", error);
    return new Response("Error", { status: 500 });
  }
}

// Foto, video, audio, documento o sticker que manda el cliente
async function procesarArchivoRecibido(message, from, paraEnviar) {
  const tipo = message.type;
  const datos = message[tipo] || {};
  const caption = datos.caption || "";
  const etiqueta = ETIQUETAS_MEDIA[tipo];

  const hiloId = await obtenerOCrearHilo(from);

  let archivo = null;
  try {

  }

  await guardarMensaje(from, hiloId, "user", mensaje, {
    tipo,
    media_url: archivo?.url || null,
    media_mime: archivo?.mime || datos.mime_type || null,
    media_nombre: datos.filename || null,
    wa_id: message.id,
  });

  await avisarNuevoMensaje(from, mensaje);

  // El bot no sabe interpretar archivos: avisa que una persona lo va a ver y se pausa
  const pausado = await estaBotPausado(from);
  if (!pausado && BOT_ACTIVO) {
    await guardarMensaje(from, hiloId, "model", AVISO_ARCHIVO_RECIBIDO);
    await mandarMensajeWhatsApp(paraEnviar, AVISO_ARCHIVO_RECIBIDO);
    await supabase
      .from("whatsapp_clientes")
      .update({ bot_pausado: true })
      .eq("numero_cliente", from);
  }
}

function extensionDeMime(mime, nombre) {
  const base = (mime || "").split(";")[0].trim().toLowerCase();
  const mapa = {
    "image/jpeg": "jpg",
    "image/png": "png",
    "image/webp": "webp",
    "video/mp4": "mp4",
    "video/3gpp": "3gp",
    "audio/ogg": "ogg",
    "audio/mpeg": "mp3",
    "audio/mp4": "m4a",
    "audio/aac": "aac",
    "audio/amr": "amr",
    "application/pdf": "pdf",
    "application/msword": "doc",
    "application/vnd.openxmlformats-officedocument.wordprocessingml.document": "docx",
    "application/vnd.ms-excel": "xls",
    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet": "xlsx",
    "application/vnd.ms-powerpoint": "ppt",
    "application/vnd.openxmlformats-officedocument.presentationml.presentation": "pptx",
    "text/plain": "txt",
  };
  if (mapa[base]) return mapa[base];

  if (nombre && nombre.includes(".")) {
    const deNombre = nombre.split(".").pop();
    if (/^[a-zA-Z0-9]{1,6}$/.test(deNombre)) return deNombre.toLowerCase();
  }
  return "bin";
}

// Baja el archivo de los servidores de Meta y lo guarda en Supabase Storage
async function descargarYGuardarArchivo(mediaId, numeroCliente, mimeDeclarado, nombreOriginal) {
  if (!mediaId) throw new Error("El mensaje no trae el id del archivo");

  const cabeceras = { Authorization: `Bearer ${process.env.WHATSAPP_TOKEN}` };

  const resInfo = await fetch(`https://graph.facebook.com/v21.0/${mediaId}`, {
    headers: cabeceras,
  });
  const info = await resInfo.json();
  if (!resInfo.ok || !info.url) {
    throw new Error("Meta no devolvió el link del archivo: " + JSON.stringify(info));
  }

  const resArchivo = await fetch(info.url, { headers: cabeceras });
  if (!resArchivo.ok) {
    throw new Error("No se pudo bajar el archivo (código " + resArchivo.status + ")");
  }
  const contenido = Buffer.from(await resArchivo.arrayBuffer());

  const mimeCompleto = info.mime_type || mimeDeclarado || resArchivo.headers.get("content-type") || "";
  const mime = mimeCompleto.split(";")[0].trim().toLowerCase() || "application/octet-stream";
  const extension = extensionDeMime(mime, nombreOriginal);
  const ruta = `entrantes/${numeroCliente}/${Date.now()}-${crypto.randomUUID()}.${extension}`;

  const { error } = await supabase.storage
    .from(BUCKET_MEDIA)
    .upload(ruta, contenido, { contentType: mime, upsert: false });
  if (error) throw new Error("No se pudo guardar en Storage: " + error.message);

  const { data } = supabase.storage.from(BUCKET_MEDIA).getPublicUrl(ruta);
  return { url: data.publicUrl, mime };
}

async function obtenerOCrearHilo(numeroCliente) {
  const { data: ultimoMensaje } = await supabase
    .from("whatsapp_conversaciones")
    .select("hilo_id, created_at")
    .eq("numero_cliente", numeroCliente)
    .order("created_at", { ascending: false })
    .limit(1)
    .maybeSingle();

  if (ultimoMensaje) {
    const fechaUltimoMensaje = new Date(ultimoMensaje.created_at).toLocaleDateString(
      "es-AR",
      { timeZone: "America/Argentina/Buenos_Aires" }
    );
    const fechaHoy = new Date().toLocaleDateString("es-AR", {
      timeZone: "America/Argentina/Buenos_Aires",
    });

    if (fechaUltimoMensaje === fechaHoy) {
      return ultimoMensaje.hilo_id; // mismo día, seguimos la misma conversación
    }
  }

  // Cambió el día: arranca un hilo nuevo
  const nuevoHiloId = crypto.randomUUID();
  return nuevoHiloId;
}

async function avisarNuevoMensaje(numeroCliente, texto) {
  const { data: suscripciones } = await supabase
    .from("push_suscripciones")
    .select("id, endpoint, p256dh, auth");

  if (!suscripciones || suscripciones.length === 0) return;

  const payload = JSON.stringify({
    titulo: "Nuevo mensaje de WhatsApp",
    cuerpo: texto.length > 100 ? texto.slice(0, 100) + "..." : texto,
    url: `/admin/conversaciones`,
  });

  await Promise.all(
    suscripciones.map(async (s) => {
      try {
        await webpush.sendNotification(
          {
            endpoint: s.endpoint,
            keys: { p256dh: s.p256dh, auth: s.auth },
          },
          payload
        );
      } catch (error) {
        if (error.statusCode === 404 || error.statusCode === 410) {
          // La suscripción ya no existe (se desinstaló la app, etc.)
          await supabase.from("push_suscripciones").delete().eq("id", s.id);
        } else {
          console.error("Error al mandar notificación push:", error.message);
        }
      }
    })
  );
}

// "confeccion" llega desde page.js con textoConfeccion(), así los días

export default function BannerCarrusel({ confeccion }) {
  // Para sumar o cambiar tarjetas, editá solo esta lista.
  // tipo 'imagen': usa un archivo de /public. tipo 'texto': título, subtítulo y detalle.
  // Si ponés href (por ejemplo '/producto/4'), la tarjeta lleva a esa página.
  const TARJETAS = [
    {
      tipo: 'imagen',
      src: '/banner-envio-gratis.png',
      alt: 'Envío gratis en compras desde $150.000 a todo el país',
      href: null,
    },
    {
      tipo: 'imagen',
      src: '/banner-descuento.png',
      alt: '5% de descuento pagando por transferencia',
      href: null,
    },
    {
      tipo: 'texto',
      titulo: 'Confección a pedido',
      subtitulo: 'Elegí tu talle y color, y lo hacemos para vos',
      detalle: `${confeccion} · Envíos a todo el país`,
      href: null,
    },
  ]

  const total = TARJETAS.length
  const [actual, setActual] = useState(0)
  const [pausado, setPausado] = useState(false)
  const x0 = useRef(null)

  function ir(n) {
    setActual((n + total) % total)
  }

  // Autoplay: se pausa con el mouse o el dedo, y se reinicia al cambiar de tarjeta
  useEffect(() => {
    if (pausado || total < 2) return
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return
    const id = setTimeout(() => setActual((a) => (a + 1) % total), INTERVALO)
    return () => clearTimeout(id)
  }, [pausado, actual, total])

  function onTouchStart(e) {
    x0.current = e.touches[0].clientX
    setPausado(true)
  }

  function onTouchEnd(e) {
    if (x0.current !== null) {
      const dx = e.changedTouches[0].clientX - x0.current
      if (Math.abs(dx) > 40) ir(actual + (dx < 0 ? 1 : -1))
    }
    x0.current = null
    setPausado(false)
  }

  function contenido(t, i) {
    if (t.tipo === 'imagen') {
      return (
        <Image
          src={t.src}
          alt={t.alt}
          width={1636}
          height={900}
          priority={i === 0}
          className="block h-full w-full object-cover"
        />
      )

        </p>
        <div
          aria-hidden="true"
          className="absolute right-[8%] top-1/2 aspect-square w-[clamp(60px,12vw,180px)] -translate-y-1/2 rounded-full bg-[#2F6B63] opacity-10"
        />
      </div>
    )
  }

  return (
    <section
      aria-roledescription="carrusel"
      aria-label="Promociones"
      className="mx-auto w-full max-w-[1200px] px-6 py-8 md:px-10"
    >

                              onClick={() => actualizarCantidad(item.id, item.variante_id, item.cantidad + 1)}
                              style={{
                                width: '28px',
                                height: '28px',
                                fontSize: '15px',
                                border: '1px solid #ccc',
                                borderRadius: '0 6px 6px 0',
                                backgroundColor: '#fff',
                                cursor: 'pointer',
                                color: '#333',
                              }}
                            >
                              +
                            </button>
                          </div>
                          <p style={{ margin: 0, fontSize: '14px', fontWeight: '600' }}>
                            ${item.cantidad * item.precio}
                          </p>
                        </div>
                      </div>
                    ))}
                  </div>

                  <div style={{ marginTop: '16px', paddingTop: '12px', borderTop: '1px solid #ddd' }}>
                    <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '14px', marginBottom: '4px' }}>
                      <span>Subtotal</span>
                      <span>${totalProductos}</span>
                    </div>
                    <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '14px', marginBottom: '4px', color: '#666' }}>
                      <span>
                        {envioGratis && zonaEnvio?.nombre !== 'Retiro en persona'
                          ? 'Costo de envío'
                          : `Envío ${zonaEnvio ? `(${zonaEnvio.nombre})` : '(sin elegir)'}`}
                      </span>
                      <span>
                        {envioGratis && zonaEnvio?.nombre !== 'Retiro en persona' ? (
                          <strong style={{ color: '#15803d' }}>¡Gratis!</strong>
                        ) : (
                          `$${costoEnvio}`
                        )}
                      </span>
                    </div>
                    <div style={{ display: 'flex', justifyContent: 'space-between', fontWeight: 'bold', fontSize: '18px', marginTop: '8px' }}>
                      <span>Total</span>
                      <span>${total}</span>
                    </div>
                  </div>

                  <div style={{ marginTop: '14px' }}>
                    <SelectorEnvio />
                  </div>

                  <Link
                    href="/checkout"
                    style={{
                      display: 'block',
                      marginTop: 'auto',
                      position: 'sticky',
                      bottom: 0,
                      backgroundColor: 'white',
                      paddingTop: '10px',
                      paddingBottom: '10px',
                      zIndex: 1,
                    }}
                  >
                    <button
                      onClick={cerrarPanel}
                      style={{
                        width: '100%',
                        marginTop: '16px',
                        padding: '12px',
                        backgroundColor: '#009ee3',
                        color: 'white',
                        border: 'none',
                        borderRadius: '6px',
                        cursor: 'pointer',
                        fontSize: '15px',
                      }}
                    >
                      Finalizar compra
                    </button>
                  </Link>
                </>
              )}
            </div>
          </>,
          document.body
        )}
    </>
  )
}

const CLAVE_CERRADO = 'dimedetiambos_cartel_suscripcion_cerrado'

export default function CartelSuscripcion() {
  const { usuario, sesionLista } = useCarrito()
  const pathname = usePathname()

  // Arranca "cerrado" hasta confirmar que la persona no lo cerró antes
  const [cerrado, setCerrado] = useState(true)
  const [demoraCumplida, setDemoraCumplida] = useState(false)

  useEffect(() => {
    try {
      setCerrado(Boolean(localStorage.getItem(CLAVE_CERRADO)))
    } catch {
      setCerrado(false)
    }
    const timer = setTimeout(() => setDemoraCumplida(true), 3000)
    return () => clearTimeout(timer)

  function cerrar() {
    try {
      localStorage.setItem(CLAVE_CERRADO, '1')

    <div

// DESTINO: src/app/checkout/page.js  (reemplaza el archivo actual)
'use client'

import Script from 'next/script'

import { useState, useEffect, useRef } from 'react'
import { useCarrito } from '@/lib/carrito'
import { useRouter } from 'next/navigation'
import SelectorEnvio from './SelectorEnvio'
import BrandBrickMP from './BrandBrickMP'
import BotonGoogle from '../BotonGoogle'
import { DESCUENTO_TRANSFERENCIA } from '@/lib/configTransferencia'
import { textoConfeccion } from '@/lib/configConfeccion'
import { ID_ENVIO_GRATIS_AUTO, UMBRAL_ENVIO_GRATIS, calcularCostoEnvio } from '@/lib/configEnvioGratis'
import { etiquetaDescuento } from '@/lib/configCupones'
import { track } from '@/lib/fbq'

export default function Checkout() {
  const {
    items,
    totalProductos,
    envioGratis,
    zonaEnvio,
    cupon,
    cuponAplica,
    descuentoCupon,
    aplicarCupon,
    quitarCupon,
    pagar,
    vaciarCarrito,
    usuario,
    cerrarSesion,
    cabecerasConSesion,
  } = useCarrito()
  const router = useRouter()

  const [datos, setDatos] = useState({
    nombre: '',
    email: '',
    dni: '',
    telefono: '',
    calle: '',
    numero: '',
    pisoDepto: '',
    ciudad: '',
    codigoPostal: '',
    notas: '',
  })

  const [enviando, setEnviando] = useState(false)
  const [metodoPago, setMetodoPago] = useState('mercadopago')

  // Campo de cupón
  const [codigoCupon, setCodigoCupon] = useState('')
  const [mensajeCupon, setMensajeCupon] = useState('')
  const [aplicandoCupon, setAplicandoCupon] = useState(false)
  const [cuponPideLogin, setCuponPideLogin] = useState(false)

  // META PIXEL: InitiateCheckout (una sola vez). Los hooks van siempre antes de cualquier return.
  const checkoutIniciado = useRef(false)

  useEffect(() => {
    if (checkoutIniciado.current || items.length === 0) return
    checkoutIniciado.current = true
    track('InitiateCheckout', {
      value: totalProductos,
      currency: 'ARS',
      content_type: 'product',
      num_items: items.reduce((a, i) => a + i.cantidad, 0),
      contents: items.map((i) => ({
        id: String(i.id),
        quantity: i.cantidad,
        item_price: i.precio,
      })),
    })
  }, [items, totalProductos])

  const esRetiro = zonaEnvio?.nombre === 'Retiro en persona'
  const porcentajeDescuento = Math.round(DESCUENTO_TRANSFERENCIA * 100)
  const conTransferencia = metodoPago === 'transferencia'

  // El cupón y el descuento por transferencia SE SUMAN: primero se descuenta el cupón
  // y el % de transferencia se calcula sobre lo que queda (5% + 5% = 9,75%).
  const hayCupon = cuponAplica && descuentoCupon > 0
  const cuponEnUso = hayCupon
  const descuentoCuponEfectivo = cuponEnUso ? descuentoCupon : 0
  const subtotalEfectivo = totalProductos - descuentoCuponEfectivo

  const montoTransferencia = Math.round(subtotalEfectivo * DESCUENTO_TRANSFERENCIA)
  const transferenciaGana = conTransferencia
  const descuentoTransferenciaEfectivo = conTransferencia ? montoTransferencia : 0
  const envioGratisEfectivo = subtotalEfectivo >= UMBRAL_ENVIO_GRATIS
  const faltaParaEnvioGratis = Math.max(0, UMBRAL_ENVIO_GRATIS - subtotalEfectivo)
  const costoEnvioEfectivo = calcularCostoEnvio(subtotalEfectivo, zonaEnvio)
  const totalAPagar = subtotalEfectivo - descuentoTransferenciaEfectivo + costoEnvioEfectivo

  function handleChange(e) {
    setDatos({ ...datos, [e.target.name]: e.target.value })
  }

  async function handleAplicarCupon() {
    setMensajeCupon('')
    setCuponPideLogin(false)
    setAplicandoCupon(true)
    const resultado = await aplicarCupon(codigoCupon)
    setAplicandoCupon(false)

    if (!resultado.ok) {
      setMensajeCupon(resultado.error)
      if (resultado.loginRequerido) setCuponPideLogin(true)
      return
    }
    setCodigoCupon('')
  }

  function validar() {
    if (cupon && !usuario) {
      return 'Para usar el cupón ingresá con Google (o quitá el cupón para seguir sin cuenta).'
    }
    if (!datos.nombre.trim()) return 'Ingresá tu nombre completo'
    if (!datos.email.trim()) return 'Ingresá tu email'
    if (!datos.dni.trim()) return 'Ingresá tu DNI'
    if (!datos.telefono.trim()) return 'Ingresá tu teléfono'
    if (envioGratis && !esRetiro) {
      // Con envío gratis la empresa se elige sola al cargar el código postal
      if (zonaEnvio?.id !== ID_ENVIO_GRATIS_AUTO) {
        return 'Todavía no pudimos confirmar tu envío gratis. Revisá tu código postal.'
      }
    } else if (!zonaEnvio) {
      return 'Elegí una zona de envío'
    }
    if (!esRetiro) {
      if (!datos.calle.trim() || !datos.numero.trim() || !datos.ciudad.trim() || !datos.codigoPostal.trim()) {
        return 'Completá la dirección de envío completa'
      }
    }
    return null
  }

  async function handleSubmit(e) {
    e.preventDefault()

    if (items.length === 0) {
      alert('Tu carrito está vacío')
      return
    }

    const error = validar()
    if (error) {
      alert(error)
      return
    }

    setEnviando(true)
    try {
      const token = await new Promise((resolve, reject) => {
        window.grecaptcha.ready(() => {
          window.grecaptcha
            .execute(process.env.NEXT_PUBLIC_RECAPTCHA_SITE_KEY, { action: 'checkout' })
            .then(resolve)
            .catch(reject)
        })
      })

      if (conTransferencia) {
        const res = await fetch('/api/crear-pedido-transferencia', {
          method: 'POST',
          headers: await cabecerasConSesion(),
          body: JSON.stringify({
            items,
            zonaEnvio,
            datosCliente: datos,
            recaptchaToken: token,
            cuponCodigo: cupon?.codigo || null,
          }),
        })
        const data = await res.json()

        if (!res.ok) {
          // Si el cupón ya no sirve (venció, se agotó o ya lo usaste), se saca para que el cliente vea el precio real
          if (data.cupon_invalido) quitarCupon()
          alert(data.error || 'Hubo un error al procesar tu pedido. Probá de nuevo.')
          setEnviando(false)
          return
        }

        // La pantalla siguiente lee estos datos para mostrar el monto y el número de pedido
        sessionStorage.setItem(
          'pedido_transferencia',
          JSON.stringify({ id: data.pedido_id, total: data.total, descuento: data.descuento })
        )

        // META PIXEL: Purchase por transferencia
        track(
          'Purchase',
          {
            value: Number(data.total),
            currency: 'ARS',
            content_type: 'product',
            num_items: items.reduce((a, i) => a + i.cantidad, 0),
            contents: items.map((i) => ({
              id: String(i.id),
              quantity: i.cantidad,
              item_price: i.precio,
            })),
          },
          `purchase_${data.pedido_id}`
        )

        vaciarCarrito()
        router.push('/pago-transferencia')
        return
      }

      await pagar(datos, token)
    } catch (err) {
      if (err?.cuponInvalido) {
        quitarCupon()
        alert(err.message)
      } else {
        alert('Hubo un error al procesar tu pedido. Probá de nuevo.')
      }
      setEnviando(false)
    }
  }

  if (items.length === 0) {
    return (
      <main style={{ padding: '40px', textAlign: 'center' }}>
        <h1>Tu carrito está vacío</h1>
        <button onClick={() => router.push('/')} style={{ marginTop: '20px', padding: '10px 20px', cursor: 'pointer' }}>
          Volver a la tienda
        </button>
      </main>
    )
  }

  return (
    <main style={{ padding: 'clamp(16px, 5vw, 40px)', maxWidth: '600px', margin: '0 auto' }}>
      <Script src={`https://www.google.com/recaptcha/api.js?render=${process.env.NEXT_PUBLIC_RECAPTCHA_SITE_KEY}`} />
      <h1>Finalizar compra</h1>
      <div style={{ backgroundColor: '#f7f7f7', padding: '16px', borderRadius: '8px', marginTop: '20px' }}>
        <h3>Resumen del pedido</h3>
        {items.map((item) => (
          <p key={`${item.id}-${item.variante_id || ''}`}>
            {item.nombre} {item.color ? `(${item.color})` : ''} x{item.cantidad} — ${item.precio * item.cantidad}
          </p>
        ))}
        {cuponEnUso && (
          <p style={{ color: '#15803d' }}>
            Cupón {cupon.codigo} ({etiquetaDescuento(cupon)}): −${descuentoCuponEfectivo}
          </p>
        )}
        {hayCupon && conTransferencia && (
          <p style={{ fontSize: '14px', color: '#15803d' }}>
            ¡Tu cupón {cupon.codigo} se suma al descuento por transferencia!
          </p>
        )}
        {envioGratisEfectivo && !esRetiro ? (
          <p>
            Costo de envío: <strong style={{ color: '#15803d' }}>¡Gratis!</strong>
          </p>
        ) : (
          <p>
            Envío ({zonaEnvio?.nombre || 'sin elegir'}): ${costoEnvioEfectivo}
          </p>
        )}
        {!esRetiro && !envioGratisEfectivo && (
          <p style={{ fontSize: '14px', color: '#15803d' }}>
            🚚 Sumá ${faltaParaEnvioGratis} más en productos y el envío es gratis.
          </p>
        )}
        {transferenciaGana && (
          <p style={{ color: '#15803d' }}>
            Descuento por transferencia ({porcentajeDescuento}%): −${descuentoTransferenciaEfectivo}
          </p>
        )}
        <p style={{ fontSize: '14px', color: '#555' }}>
          🧵 Tiempo de confección estimado: {textoConfeccion()}.
          {esRetiro ? ' Te avisamos cuando esté listo para retirar.' : ''}
        </p>
        <p style={{ fontWeight: 'bold', fontSize: '18px' }}>Total: ${totalAPagar}</p>
      </div>

      {/* Ingreso con Google: hace falta para usar cupones */}
      <div style={{ marginTop: '16px' }}>
        {!usuario ? (
          <div
            style={{
              padding: '14px',
              border: '1px solid #E4DCCF',
              backgroundColor: '#FAF6F0',
              borderRadius: '8px',
            }}
          >
            <p style={{ margin: '0 0 10px', fontSize: '14px', color: '#2E2A26' }}>
              🎁 Ingresá con Google para usar cupones y llevarte <strong>5% de descuento en tu primera compra</strong>.
              Podés comprar sin cuenta, pero sin cupón.
            </p>
            <BotonGoogle />
            <p style={{ margin: '10px 0 0', fontSize: '12px', color: '#8A8378' }}>
              Al ingresar aceptás recibir novedades por email. Podés darte de baja cuando quieras.
            </p>
          </div>
        ) : (
          <p style={{ margin: 0, fontSize: '13px', color: '#555' }}>
            Ingresaste como <strong>{usuario.email}</strong> ·{' '}
            <button
              type="button"
              onClick={cerrarSesion}
              style={{ border: 'none', background: 'none', color: '#2F6B63', cursor: 'pointer', textDecoration: 'underline', padding: 0 }}
            >
              Salir
            </button>
          </p>
        )}
      </div>

      <div style={{ marginTop: '16px' }}>
        {cupon ? (
          <div
            style={{
              display: 'flex',
              justifyContent: 'space-between',
              alignItems: 'center',
              gap: '10px',
              padding: '12px',
              border: '1px solid #bbf7d0',
              backgroundColor: '#f0fdf4',
              borderRadius: '8px',
            }}
          >
            <span style={{ fontSize: '14px', color: '#15803d' }}>
              🎟️ Cupón <strong>{cupon.codigo}</strong> ({etiquetaDescuento(cupon)})
              {!cuponAplica && (
                <>
                  <br />
                  <small style={{ color: '#a16207' }}>
                    Se aplica con una compra mínima de ${Number(cupon.compra_minima).toLocaleString('es-AR')} en productos.
                  </small>
                </>
              )}

            <button
              type="button"
              onClick={quitarCupon}
              style={{ padding: '6px 12px', border: '1px solid #ccc', borderRadius: '6px', background: '#fff', cursor: 'pointer' }}
            >
              Quitar
            </button>
          </div>
        ) : (
          <div style={{ display: 'flex', gap: '8px' }}>
            <input
              value={codigoCupon}
              onChange={(e) => setCodigoCupon(e.target.value)}
              onKeyDown={(e) => {
                if (e.key === 'Enter') {
                  e.preventDefault()
                  if (codigoCupon.trim() && !aplicandoCupon) handleAplicarCupon()
                }
              }}
              placeholder="¿Tenés un cupón?"
              autoCapitalize="characters"
              style={{ ...inputStyle, marginTop: 0, flex: 1 }}
            />
            <button
              type="button"
              onClick={handleAplicarCupon}
              disabled={aplicandoCupon || !codigoCupon.trim()}
              style={{
                padding: '0 18px',
                border: 'none',
                borderRadius: '6px',
                color: 'white',
                backgroundColor: aplicandoCupon || !codigoCupon.trim() ? '#ccc' : '#2F6B63',
                cursor: aplicandoCupon || !codigoCupon.trim() ? 'not-allowed' : 'pointer',
              }}
            >
              {aplicandoCupon ? '...' : 'Aplicar'}
            </button>
          </div>
        )}
        {mensajeCupon && <p style={{ margin: '8px 0 0', fontSize: '14px', color: '#b91c1c' }}>{mensajeCupon}</p>}
        {cuponPideLogin && !usuario && (
          <div style={{ marginTop: '10px' }}>
            <BotonGoogle texto="Ingresar con Google para usar el cupón" />
          </div>
        )}
      </div>

      <SelectorEnvio codigoPostal={datos.codigoPostal} />

      <form onSubmit={handleSubmit} style={{ marginTop: '20px', display: 'flex', flexDirection: 'column', gap: '12px' }}>
        <label>
          Nombre completo *

          Teléfono *
          <input name="telefono" value={datos.telefono} onChange={handleChange} style={inputStyle} />

        {!esRetiro && (

                <input name="codigoPostal" value={datos.codigoPostal} onChange={handleChange} style={inputStyle} />

        <label>
          Notas adicionales (opcional)
          <textarea name="notas" value={datos.notas} onChange={handleChange} style={{ ...inputStyle, minHeight: '60px' }} />
        </label>

        <div style={{ marginTop: '8px' }}>
          <p style={{ margin: '0 0 8px', fontWeight: 'bold' }}>Forma de pago</p>

        </button>
      </form>
    </main>
  )
}

const inputStyle = {

  return min === max ? `${min}` : `${min} a ${max}`
}

// Plazos por separado: total, confección (de configConfeccion.js) y envío (según la zona, los define la cotización)

  const [cpInput, setCpInput] = useState('')
  const [opcionesEnvia, setOpcionesEnvia] = useState([])
  const [cargando, setCargando] = useState(false)
  const [errorCotizacion, setErrorCotizacion] = useState(null)
  
  const { items, zonaEnvio, setZonaEnvio, envioGratis } = useCarrito()

  const retiroSeleccionado = zonaEnvio?.id === RETIRO_EN_PERSONA.id

  // En el checkout se le pasa el código postal del formulario; en el panel del carrito no
  const dentroDelCheckout = cpDireccion !== undefined

  // Sincronizar si el usuario cambia el CP en el formulario principal
  useEffect(() => {
    if (cpDireccion && cpDireccion !== cpInput) {
      setCpInput(cpDireccion)
    }
  }, [cpDireccion])

  // Función para consultar la cotización a Envia.com
  const cotizar = async (cp) => {
    if (!cp || String(cp).trim().length < 4) {
      setErrorCotizacion('Ingresá un Código Postal válido (4 dígitos).')
      setOpcionesEnvia([])
      return
    }

    setCargando(true)
    setErrorCotizacion(null)

    try {
      const res = await fetch('/api/envios/cotizar', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          postalCode: String(cp).trim(),
          items: items,
        }),
      })

      const data = await res.json()

      if (res.ok && data.success && data.rates?.length > 0) {
        setOpcionesEnvia(data.rates)
        setErrorCotizacion(null)
      } else {
        setErrorCotizacion('No se encontraron opciones de envío para este Código Postal.')
        setOpcionesEnvia([])
      }
    } catch (err) {
      setErrorCotizacion('Error al calcular el envío. Intentalo de nuevo.')
      setOpcionesEnvia([])
    } finally {
      setCargando(false)
    }
  }

  // Cotizar automáticamente si el CP tiene 4 dígitos
  useEffect(() => {
    if (cpInput.trim().length >= 4) {
      const timer = setTimeout(() => cotizar(cpInput), 500)

    }

'use client'

import { useEffect, useState } from 'react'
import { useCarrito } from '@/lib/carrito'
import { etiquetaDescuento } from '@/lib/configCupones'

// Si el cliente entra con un link como https://www.dimedetiambos.com.ar/?cupon=VERANO10
// (por ejemplo, escaneando un QR), el cupón queda cargado solo y se muestra un aviso.
export default function CuponDesdeQR() {
  const { aplicarCupon } = useCarrito()
  const [aviso, setAviso] = useState(null)

  useEffect(() => {
    const params = new URLSearchParams(window.location.search)
    const codigo = params.get('cupon')
    if (!codigo) return

    // Se saca el cupón de la dirección para que no se vuelva a aplicar al recargar
    params.delete('cupon')
    const resto = params.toString()
    window.history.replaceState(
      null,
      '',
      window.location.pathname + (resto ? `?${resto}` : '') + window.location.hash
    )

    aplicarCupon(codigo).then((r) => {
      if (r.ok) {
        const minimo = Number(r.cupon.compra_minima) || 0
        setAviso({
          ok: true,
          texto:
            `🎟️ Cupón ${r.cupon.codigo} aplicado (${etiquetaDescuento(r.cupon)})` +
            (minimo > 0 ? ` · compra mínima $${minimo.toLocaleString('es-AR')}` : ''),
        })
      } else {
        setAviso({ ok: false, texto: r.error })
      }
    })
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  useEffect(() => {
    if (!aviso) return
    const timer = setTimeout(() => setAviso(null), 7000)
    return () => clearTimeout(timer)
  }, [aviso])

  if (!aviso) return null

  return (
    <div
      role="status"
      onClick={() => setAviso(null)}
      style={{
        position: 'fixed',
        top: '12px',
        left: '50%',
        transform: 'translateX(-50%)',
        zIndex: 1000,
        maxWidth: 'calc(100vw - 24px)',
        padding: '12px 18px',
        borderRadius: '10px',
        fontSize: '14px',
        textAlign: 'center',
        cursor: 'pointer',
        color: 'white',
        backgroundColor: aviso.ok ? '#2F6B63' : '#b91c1c',
        boxShadow: '0 4px 14px rgba(0,0,0,0.2)',
      }}
    >
      {aviso.texto}
    </div>
  )
}

const UMBRAL_AGRANDAR = 10

  useEffect(() => {

    }
    onScroll()
    window.addEventListener('scroll', onScroll, { passive: true })

// Orden del catálogo: primero ambos, luego chaquetas y al final pantalones.
// Se mira cómo EMPIEZA el nombre, para que "Ambo ... Pantalon con Puños" no caiga en pantalones.
function grupo(nombre = '') {
  const n = nombre
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .trim()

  if (n.startsWith('ambo')) return 0
  if (n.startsWith('chaqueta')) return 1
  if (n.startsWith('pantalon')) return 2
  return 3
}

  }

  }

  function alternarPanel() {
    setPanelAbierto((abierto) => !abierto)
  }

  // Al montar en el navegador, recuperar lo guardado

      const itemsGuardados = localStorage.getItem(CLAVE_ITEMS)

      setUsuario(sesion?.user ?? null)
      setSesionLista(true)
    })

    return () => {
      activo = false
      suscripcion.subscription.unsubscribe()
    }
  }, [])

  // Guardar items cada vez que cambian (una vez que ya se cargó lo inicial)
  useEffect(() => {
    if (!cargado) return
    try {
      localStorage.setItem(CLAVE_ITEMS, JSON.stringify(items))
    } catch (err) {
      console.error('Error al guardar el carrito:', err)
    }
  }, [items, cargado])

  // Guardar zona de envío cada vez que cambia
  useEffect(() => {
    if (!cargado) return
    try {
      if (zonaEnvio) {
        localStorage.setItem(CLAVE_ZONA, JSON.stringify(zonaEnvio))
      } else {
        localStorage.removeItem(CLAVE_ZONA)
      }
    } catch (err) {
      console.error('Error al guardar la zona de envío:', err)
    }
  }, [zonaEnvio, cargado])

  // Guardar el cupón cada vez que cambia
  useEffect(() => {
    if (!cargado) return
    try {
      if (cupon) {
        localStorage.setItem(CLAVE_CUPON, JSON.stringify(cupon))
      } else {
        localStorage.removeItem(CLAVE_CUPON)
      }
    } catch (err) {
      console.error('Error al guardar el cupón:', err)
    }
  }, [cupon, cargado])

  function agregarAlCarrito(producto, cantidad = 1) {
    setItems((prev) => {
      const existe = prev.find(
        (item) => item.id === producto.id && item.variante_id === producto.variante_id
      )
      if (existe) {
        return prev.map((item) =>
          item.id === producto.id && item.variante_id === producto.variante_id
            ? { ...item, cantidad: item.cantidad + cantidad }
            : item
        )
      }
      return [...prev, { ...producto, cantidad }]
    })
  }

  function quitarDelCarrito(id, varianteId) {
    setItems((prev) =>
      prev.filter((item) => !(item.id === id && item.variante_id === varianteId))
    )
  }

  function actualizarCantidad(id, varianteId, nuevaCantidad) {
    if (nuevaCantidad < 1) {
      quitarDelCarrito(id, varianteId)
      return
    }
    setItems((prev) =>
      prev.map((item) =>
        item.id === id && item.variante_id === varianteId
          ? { ...item, cantidad: nuevaCantidad }
          : item
      )
    )
  }

  function vaciarCarrito() {
    setItems([])
    setZonaEnvio(null)
    setCupon(null)
  }

  // ---------- Sesión con Google ----------

  async function iniciarSesionGoogle() {
    const volverA = window.location.href.split('#')[0]
    await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: { redirectTo: volverA },
    })
  }

  async function cerrarSesion() {
    await supabase.auth.signOut()
    setCupon(null)
  }

  // Cabeceras para las llamadas a la API: incluyen la sesión para que el servidor sepa quién es
  async function cabecerasConSesion() {
    const cabeceras = { 'Content-Type': 'application/json' }
    try {
      const { data } = await supabase.auth.getSession()
      const token = data.session?.access_token
      if (token) cabeceras.Authorization = `Bearer ${token}`
    } catch (err) {
      console.error('No se pudo leer la sesión:', err)
    }
    return cabeceras
  }

  // Comprueba con el servidor que el cupón exista y esté vigente, y lo deja cargado.
  // Para usar un cupón hay que haber ingresado con Google: si no, se recuerda el código
  // y se aplica solo apenas la persona ingresa.
  // El descuento definitivo se vuelve a calcular en el servidor al pagar.
  async function aplicarCupon(codigo) {
    try {
      const respuesta = await fetch('/api/validar-cupon', {
        method: 'POST',
        headers: await cabecerasConSesion(),
        body: JSON.stringify({ codigo }),
      })
      const datos = await respuesta.json()

      if (respuesta.status === 401 || datos.login_requerido) {
        try {
          localStorage.setItem(CLAVE_CUPON_PENDIENTE, normalizarCodigo(codigo))
        } catch {}
        return {
          ok: false,
          loginRequerido: true,
          error: datos.error || 'Para usar un cupón ingresá con Google.',
        }
      }

      if (!respuesta.ok || !datos.cupon) {
        return { ok: false, error: datos.error || 'No pudimos aplicar el cupón.' }
      }

      setCupon(datos.cupon)
      return { ok: true, cupon: datos.cupon }
    } catch (err) {
      return { ok: false, errorDeRed: true, error: 'No pudimos verificar el cupón. Probá de nuevo.' }
    }
  }

  function quitarCupon() {
    setCupon(null)
  }

  // Apenas la persona ingresa con Google:
  // 1) si había un cupón esperando (ej: venía de un QR), se aplica
  // 2) si no hay ningún cupón puesto, se prueba el de bienvenida (si ya lo usó, no pasa nada)
  useEffect(() => {
    if (!cargado || !sesionLista || !usuario) return

    let pendiente = null
    try {
      pendiente = localStorage.getItem(CLAVE_CUPON_PENDIENTE)
      if (pendiente) localStorage.removeItem(CLAVE_CUPON_PENDIENTE)
    } catch {}

    if (pendiente) {
      aplicarCupon(pendiente)
      return
    }

    if (!cupon) {
      const claveProbada = `${CLAVE_BIENVENIDA_PROBADA}_${usuario.id}`
      let yaProbada = false
      try {
        yaProbada = Boolean(sessionStorage.getItem(claveProbada))
      } catch {}
      if (!yaProbada) {
        aplicarCupon(CODIGO_BIENVENIDA).then((resultado) => {
          // Se marca como probada solo si el servidor contestó de verdad (lo aplicó o lo rechazó).
          // Si falló por red o por falta de sesión, se reintenta la próxima vez.
          if (resultado.ok || (!resultado.loginRequerido && !resultado.errorDeRed)) {
            try {
              sessionStorage.setItem(claveProbada, '1')
            } catch {}
          }
        })
      }
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [cargado, sesionLista, usuario])

  async function pagar(datosCliente, recaptchaToken) {
    const respuesta = await fetch('/api/crear-preferencia', {
      method: 'POST',
      headers: await cabecerasConSesion(),
      body: JSON.stringify({
        items,
        zonaEnvio,
        datosCliente,
        recaptchaToken,
        cuponCodigo: cupon?.codigo || null,
      }),
    })

    const datos = await respuesta.json()

    if (datos.init_point) {
      window.location.href = datos.init_point
    } else {
      const error = new Error(datos.error || 'No se pudo generar el link de pago')
      error.cuponInvalido = Boolean(datos.cupon_invalido)
      throw error
    }
  }

  const totalProductos = items.reduce((acc, item) => acc + item.precio * item.cantidad, 0)

  // Cupón: descuenta solo los productos y solo si se llega a la compra mínima
  const cuponAplica = Boolean(cupon) && totalProductos >= (Number(cupon.compra_minima) || 0)
  const descuentoCupon = cuponAplica ? calcularDescuentoCupon(cupon, totalProductos) : 0
  const totalProductosConCupon = totalProductos - descuentoCupon

  // Envío gratis desde cierto monto de productos, medido DESPUÉS del cupón (ver configEnvioGratis.js)
  const envioGratis = totalProductosConCupon >= UMBRAL_ENVIO_GRATIS
  const faltaParaEnvioGratis = Math.max(0, UMBRAL_ENVIO_GRATIS - totalProductosConCupon)
  const costoEnvio = calcularCostoEnvio(totalProductosConCupon, zonaEnvio)
  const total = totalProductosConCupon + costoEnvio
  const cantidadTotal = items.reduce((acc, item) => acc + item.cantidad, 0)

  return (
    <CarritoContext.Provider
      value={{
        items,
        agregarAlCarrito,
        quitarDelCarrito,
        actualizarCantidad,
        vaciarCarrito,
        total,
        totalProductos,
        totalProductosConCupon,
        costoEnvio,
        envioGratis,
        faltaParaEnvioGratis,
        cantidadTotal,
        pagar,
        zonaEnvio,
        setZonaEnvio,
        cupon,
        cuponAplica,
        descuentoCupon,
        aplicarCupon,
        quitarCupon,
        usuario,
        sesionLista,
        iniciarSesionGoogle,
        cerrarSesion,
        cabecerasConSesion,
        panelAbierto,
        abrirPanel,
        cerrarPanel,
        alternarPanel,
      }}
    >
      {children}
    </CarritoContext.Provider>
  )
}

export function useCarrito() {
  return useContext(CarritoContext)
}

// Archivo: src/lib/configCupones.js
// Funciones puras (sin base de datos): las usan tanto el navegador como el servidor.
//
// Reglas de los cupones:
// - Descuentan solo los productos, no el envío.
// - Se aplican ANTES del 5% de transferencia (el 5% se calcula sobre el precio ya con cupón).
// - El monto del envío gratis se mide DESPUÉS del cupón.

export function normalizarCodigo(texto) {
  return String(texto || '').trim().toUpperCase().replace(/\s+/g, '')
}

// Cuánto descuenta el cupón sobre el subtotal de productos. Nunca más que el subtotal.
export function calcularDescuentoCupon(cupon, subtotal) {
  if (!cupon || !(subtotal > 0)) return 0
  const valor = Number(cupon.valor) || 0
  const bruto = cupon.tipo === 'porcentaje' ? Math.round((subtotal * valor) / 100) : Math.round(valor)
  return Math.max(0, Math.min(bruto, subtotal))
}

export function etiquetaDescuento(cupon) {
  if (cupon.tipo === 'porcentaje') return `−${Number(cupon.valor)}%`
  return `−$${Number(cupon.valor).toLocaleString('es-AR')}`
}

// Texto de la línea que queda en el pedido, el admin y los emails
export function nombreLineaCupon(cupon) {
  return `Cupón ${cupon.codigo} (${etiquetaDescuento(cupon)})`
}

// Envío gratis a partir de cierto monto.
// Archivo: src/lib/configEnvioGratis.js
//
// El monto se compara contra el total de los PRODUCTOS (antes del descuento por transferencia
// y sin contar el envío). Para cambiar el monto, editá UMBRAL_ENVIO_GRATIS y hacé git push.

export const UMBRAL_ENVIO_GRATIS = 150000

// Cuando el envío es gratis, el cliente NO elige empresa: el sistema la elige solo.
// Orden de preferencia (siempre a domicilio). Se usa la primera que tenga tarifa para su código postal.
export const EMPRESAS_ENVIO_GRATIS = ['urbano', 'oca']

// Dentro de cada empresa se prefiere este servicio (OCA lo llama "Sucursal a puerta").
// Si no aparece, se usa el más barato de los que llegan a la puerta del cliente.
export const SERVICIOS_PREFERIDOS = ['sucursal a puerta']

// Si ninguna de las anteriores llega a ese código postal, se usa la a domicilio más barata de
// cualquier otra empresa, para no perder la venta. Poné false si preferís que no se pueda comprar.
export const USAR_OTRAS_EMPRESAS_SI_FALTAN = true

// Lo que se guarda como "zona de envío" cuando el envío es gratis (el cliente no ve la empresa)
export const ID_ENVIO_GRATIS_AUTO = 'envio-gratis-auto'
export const NOMBRE_ENVIO_GRATIS = 'Envío gratis a domicilio'

// Devuelve el costo de envío que realmente se cobra
export function calcularCostoEnvio(subtotalProductos, zonaEnvio) {
  if (!zonaEnvio) return 0
  if (subtotalProductos >= UMBRAL_ENVIO_GRATIS) return 0
  return Number(zonaEnvio.costo) || 0
}

function normalizar(texto) {
  return String(texto || '')
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
}

// Una opción es "a domicilio" si el cliente la recibe en su puerta.
// OCA: "Sucursal a puerta" y "Puerta a puerta" llegan al domicilio; "Sucursal a sucursal" y
// "Puerta a sucursal" no (el destino es una sucursal).
function esADomicilio(opcion) {
  const texto = `${normalizar(opcion.service)} ${normalizar(opcion.serviceCode)}`
  if (/\ba\s+(sucursal|agencia|punto|locker)/.test(texto)) return false
  if (/puerta|domicilio|home|door/.test(texto)) return true
  if (/sucursal|agencia|punto|retiro|pickup|locker/.test(texto)) return false
  if (opcion.branches?.length > 0) return false
  return true
}

function esServicioPreferido(opcion) {
  const texto = `${normalizar(opcion.service)} ${normalizar(opcion.serviceCode)}`
  return SERVICIOS_PREFERIDOS.some((s) => texto.includes(s))
}

// Primero el servicio preferido y, entre iguales, el más barato
function ordenar(opciones) {
  return [...opciones].sort(
    (a, b) =>
      Number(esServicioPreferido(b)) - Number(esServicioPreferido(a)) || precioDe(a) - precioDe(b)
  )
}

function precioDe(opcion) {
  return Number(opcion.price) || 0
}

// De todas las tarifas que devuelve la cotización, elige la que se usa cuando el envío es gratis
export function elegirOpcionEnvioGratis(opciones) {
  const aDomicilio = (opciones || []).filter(esADomicilio)

  for (const empresa of EMPRESAS_ENVIO_GRATIS) {
    const candidatas = aDomicilio.filter((o) =>
      `${normalizar(o.carrierSlug)} ${normalizar(o.carrier)}`.includes(empresa)
    )
    if (candidatas.length > 0) {
      return ordenar(candidatas)[0]
    }
  }

  if (USAR_OTRAS_EMPRESAS_SI_FALTAN && aDomicilio.length > 0) {
    return ordenar(aDomicilio)[0]
  }

  return null
}

import { createClient } from '@supabase/supabase-js'
import { normalizarCodigo, calcularDescuentoCupon } from '@/lib/configCupones'

// Mensaje único para todos los casos, así nadie puede averiguar qué códigos existen
const MENSAJE_NO_VALIDO = 'El cupón no es válido o ya no está disponible.'
export const MENSAJE_LOGIN = 'Para usar un cupón ingresá con Google.'
export const MENSAJE_YA_USADO = 'Ya usaste este cupón con tu cuenta.'

// Lee la sesión de Google que el navegador manda en la cabecera Authorization.
// Devuelve el usuario, o null si no hay sesión válida.
export async function usuarioDesdeRequest(request) {
  const cabecera = request.headers.get('authorization') || ''
  const token = cabecera.startsWith('Bearer ') ? cabecera.slice(7).trim() : ''
  if (!token) return null

  const { data, error } = await supabaseAdmin.auth.getUser(token)
  if (error || !data?.user) return null
  return data.user
}

// ¿Esta cuenta ya usó este cupón?
async function cuentaYaUsoCupon(userId, codigo) {
  const { data, error } = await supabaseAdmin
    .from('cupones_usos')
    .select('id')
    .eq('user_id', userId)
    .eq('codigo', codigo)
    .maybeSingle()

  if (error) {
    console.error('Error al revisar los usos del cupón:', error)
    return false
  }
  return Boolean(data)
}

// Comprueba que el cupón exista y esté vigente (sin mirar el monto de la compra).
// Si se pasa userId, también comprueba que esa cuenta no lo haya usado antes.
export async function buscarCuponValido(codigoCrudo, userId = null) {
  const codigo = normalizarCodigo(codigoCrudo)
  if (!codigo) return { ok: false, error: 'Ingresá un código de cupón.' }

  const { data: cupon, error } = await supabaseAdmin
    .from('cupones')
    .select('*')
    .eq('codigo', codigo)
    .maybeSingle()

  if (error) {
    console.error('Error al buscar el cupón:', error)
    return { ok: false, error: 'No pudimos verificar el cupón. Probá de nuevo.' }
  }

  if (!cupon || !cupon.activo) return { ok: false, error: MENSAJE_NO_VALIDO }

  const ahora = new Date()
  if (cupon.fecha_inicio && new Date(cupon.fecha_inicio) > ahora) return { ok: false, error: MENSAJE_NO_VALIDO }
  if (cupon.fecha_fin && new Date(cupon.fecha_fin) < ahora) return { ok: false, error: MENSAJE_NO_VALIDO }
  if (cupon.max_usos != null && cupon.usos >= cupon.max_usos) return { ok: false, error: MENSAJE_NO_VALIDO }

  if (userId && (await cuentaYaUsoCupon(userId, cupon.codigo))) {
    return { ok: false, error: MENSAJE_YA_USADO }
  }

  return { ok: true, cupon }
}

// Valida el cupón contra el subtotal real del pedido y calcula el descuento
export async function validarCuponParaPedido(codigo, subtotal, userId = null) {
  const resultado = await buscarCuponValido(codigo, userId)
  if (!resultado.ok) return resultado

  const minimo = Number(resultado.cupon.compra_minima) || 0
  if (subtotal < minimo) {
    return {
      ok: false,
      error: `Este cupón requiere una compra mínima de $${minimo.toLocaleString('es-AR')} en productos.`,
    }
  }

  const descuento = calcularDescuentoCupon(resultado.cupon, subtotal)
  if (descuento <= 0) return { ok: false, error: MENSAJE_NO_VALIDO }

  return { ok: true, cupon: resultado.cupon, descuento }
}

  if (error) {

    return false
  }
  return data === true
}

}

// Anota que esta cuenta usó el cupón. Devuelve 'ok', 'ya_usado' o 'error'.
// El índice único de la base impide que una misma cuenta lo use dos veces,
// aunque dos pedidos lleguen al mismo tiempo.
export async function reservarUsoCuenta(userId, codigo, pedidoRef = null) {
  const { error } = await supabaseAdmin.from('cupones_usos').insert({
    user_id: userId,
    codigo,
    pedido_id: pedidoRef != null ? String(pedidoRef) : null,
  })

  if (!error) return 'ok'
  if (error.code === '23505') return 'ya_usado'
  console.error('Error al anotar el uso del cupón por cuenta:', error)

}

    .delete()
    .eq('user_id', userId)
    .eq('codigo', codigo)

}

// Mercado Pago no acepta líneas con precio negativo: el descuento se reparte entre los
// precios de los productos, proporcionalmente, y la suma queda igual al total con cupón.
export function repartirDescuentoEnItems(items, descuento) {
  if (!descuento || descuento <= 0) return items

  const subtotal = items.reduce((acc, i) => acc + i.unit_price * i.quantity, 0)
  if (subtotal <= 0) return items

  let restante = descuento
  return items.map((item, idx) => {
    const totalLinea = item.unit_price * item.quantity
    const esUltima = idx === items.length - 1
    let descLinea = esUltima ? restante : Math.round((descuento * totalLinea) / subtotal)
    descLinea = Math.max(0, Math.min(descLinea, totalLinea, restante))
    restante -= descLinea
    const nuevoPrecio = Math.round(((totalLinea - descLinea) / item.quantity) * 100) / 100
    return { ...item, unit_price: nuevoPrecio }
  })
}

      const subtotal = item.precio * item.cantidad
      const importe = subtotal < 0 ? `-$${Math.abs(subtotal)}` : `$${subtotal}`
      const detalle = subtotal < 0 ? importe : `Cantidad: ${item.cantidad} — ${importe}`

      return `
        <table role="presentation" width="100%" style="margin-bottom: 10px;">
          <tr>
            ${foto}
            <td style="vertical-align: top; font-size: 14px; color: #2E2A26;">
              <div>${item.nombre}</div>
              <div style="color: #8A8378;">${detalle}</div>
            </td>
          </tr>
        </table>
      `
    })
    .join('')
}

// Bloque de entrega para los emails al cliente: tiempo de confección y, si corresponde, aviso de retiro
function bloqueEntrega(pedido) {
  const esRetiro = pedido.zona_envio === 'Retiro en persona'

}

   function lineaEnvio(pedido) {
     if (!pedido.zona_envio) return ''
     const esRetiro = pedido.zona_envio === 'Retiro en persona'
     const texto =
       Number(pedido.costo_envio) === 0 && !esRetiro

         : `Envío (${pedido.zona_envio}): $${pedido.costo_envio}`
     return `<p style="margin: 10px 0 0; color: #2E2A26;">${texto}</p>`
   }

function formatearDireccion(direccion) {
  if (!direccion) return ''

    `${direccion.calle} ${direccion.numero}`,

    direccion.ciudad,
    direccion.codigoPostal ? `CP ${direccion.codigoPostal}` : null,
  ].filter(Boolean)
  return partes.join(', ')
}

import { PDFDocument, rgb, StandardFonts } from 'pdf-lib'
import QRCode from 'qrcode'

export async function generarFacturaPdf({ pedido, resultado, puntoVenta, tipoComprobante, numeroComprobante }) {
  const pdfDoc = await PDFDocument.create()
  const page = pdfDoc.addPage([595, 842]) // A4
  const font = await pdfDoc.embedFont(StandardFonts.Helvetica)
  const fontBold = await pdfDoc.embedFont(StandardFonts.HelveticaBold)

  let y = 800

  const dibujarTexto = (texto, x, tamano = 11, negrita = false) => {
    page.drawText(texto, {
      x,
      y,
      size: tamano,
      font: negrita ? fontBold : font,
      color: rgb(0, 0, 0),
    })
  }

  dibujarTexto('Dimedeti Ambos', 40, 18, true)
  y -= 20
  dibujarTexto('FACTURA C', 40, 14, true)
  y -= 20
  dibujarTexto(`Punto de Venta: ${String(puntoVenta).padStart(4, '0')}  Comp. Nro: ${String(numeroComprobante).padStart(8, '0')}`, 40)
  y -= 40

  dibujarTexto(`Fecha: ${new Date().toLocaleDateString('es-AR')}`, 40)
  y -= 20
  dibujarTexto(`CUIT: ${process.env.AFIP_CUIT}`, 40)
  y -= 20
  dibujarTexto('Condición frente al IVA: Monotributo', 40)
  y -= 40

  dibujarTexto(`Cliente: ${pedido.cliente_nombre || 'Consumidor Final'}`, 40)
  y -= 20
  if (pedido.cliente_dni) {
    dibujarTexto(`DNI: ${pedido.cliente_dni}`, 40)
    y -= 20
  }
  y -= 20

  dibujarTexto('Detalle:', 40, 12, true)
  y -= 20
  for (const item of pedido.items || []) {
    if (item.nombre?.toLowerCase().startsWith('envío')) continue

    // Una línea con precio negativo es un descuento (ej: pago por transferencia)
    const importe = item.precio * item.cantidad
    const linea =
      importe < 0
        ? `${item.nombre} — -$${Math.abs(importe)}`
        : `${item.cantidad} x ${item.nombre} — $${item.precio}`

    dibujarTexto(linea, 50)
    y -= 18
  }
  y -= 20

  dibujarTexto(`TOTAL: $${pedido.total}`, 40, 14, true)
  y -= 50

  dibujarTexto(`CAE: ${resultado.CAE}`, 40)
  y -= 20
  dibujarTexto(`Vencimiento CAE: ${resultado.CAEFchVto}`, 40)
  y -= 60

  // QR fiscal obligatorio (RG 4291)
  const qrPayload = {
    ver: 1,
    fecha: new Date().toISOString().slice(0, 10),
    cuit: Number(process.env.AFIP_CUIT),
    ptoVta: puntoVenta,
    tipoCmp: tipoComprobante,
    nroCmp: numeroComprobante,
    importe: Number(pedido.total),
    moneda: 'PES',
    ctz: 1,
    tipoDocRec: pedido.cliente_dni ? 96 : 99,
    nroDocRec: pedido.cliente_dni ? Number(pedido.cliente_dni.replace(/\D/g, '')) : 0,
    tipoCodAut: 'E',
    codAut: Number(resultado.CAE),
  }

  const qrUrl = `https://www.afip.gob.ar/fe/qr/?p=${Buffer.from(JSON.stringify(qrPayload)).toString('base64')}`
  const qrDataUrl = await QRCode.toDataURL(qrUrl)
  const qrImageBytes = Buffer.from(qrDataUrl.split(',')[1], 'base64')
  const qrImage = await pdfDoc.embedPng(qrImageBytes)

  page.drawImage(qrImage, { x: 40, y: y - 100, width: 100, height: 100 })

  return await pdfDoc.save()
}

import { MercadoPagoConfig, Payment } from 'mercadopago'
import { createClient } from '@supabase/supabase-js'
import { enviarEmailConfirmacionCliente, enviarEmailNotificacionVenta } from './enviarEmails'
import { generarFactura } from './generarFactura'
import { reservarUsoCuenta } from './cupones'

const client = new MercadoPagoConfig({
  accessToken: process.env.MERCADOPAGO_ACCESS_TOKEN,
})

const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY
)

export async function guardarPedido(paymentId) {
  const payment = new Payment(client)
  const pago = await payment.get({ id: paymentId })

  const itemsOriginales = pago.metadata?.items_originales || []

  const items = pago.additional_info?.items?.map((item, idx) => ({
    nombre: item.title,
    cantidad: item.quantity,
    precio: item.unit_price,
    imagen_url: itemsOriginales[idx]?.imagen_url || null,
  })) || []

  // Ver si este pedido ya existía antes con estado aprobado (para no reprocesar/duplicar emails)
  const { data: pedidoExistente } = await supabaseAdmin
    .from('pedidos')
    .select('estado')
    .eq('mercadopago_payment_id', String(paymentId))
    .maybeSingle()

  const yaEstabaAprobado = pedidoExistente?.estado === 'approved'

  const datosCliente = pago.metadata?.datos_cliente || {}
  const zonaEnvio = pago.metadata?.zona_envio || null

  const { error } = await supabaseAdmin
    .from('pedidos')
    .upsert(
      {
        mercadopago_payment_id: String(paymentId),
        estado: pago.status,
        total: pago.transaction_amount,
        items: items,
        comprador_email: datosCliente.email || pago.payer?.email || null,
        cliente_nombre: datosCliente.nombre || null,
        cliente_dni: datosCliente.dni || null,
        cliente_telefono: datosCliente.telefono || null,
        direccion: datosCliente.calle
          ? {
              calle: datosCliente.calle,
              numero: datosCliente.numero,
              pisoDepto: datosCliente.pisoDepto || datosCliente.piso_depto || null,
              ciudad: datosCliente.ciudad,
              codigoPostal: datosCliente.codigoPostal || datosCliente.codigo_postal || null,
            }
          : null,
        zona_envio: zonaEnvio?.nombre || null,
        costo_envio: zonaEnvio?.costo || 0,
        envio_carrier: zonaEnvio?.envioCarrier || zonaEnvio?.envio_carrier || null,
        envio_service_code: zonaEnvio?.envioServiceCode || zonaEnvio?.envio_service_code || null,
        envio_branch_code: zonaEnvio?.envioBranchCode || zonaEnvio?.envio_branch_code || null,
        notas: datosCliente.notas || null,
      },
      { onConflict: 'mercadopago_payment_id' }
    )

  if (error) {
    console.error('Error al guardar el pedido:', error)
    throw new Error('Error al guardar el pedido')
  }

  // Pago aprobado por primera vez: si llevaba cupón, se anota que esa cuenta ya lo usó
  // (si el pago nunca se aprobaba, la persona podía volver a intentar con el mismo cupón)
  if (pago.status === 'approved' && !yaEstabaAprobado) {
    const cuponUsado = pago.metadata?.cupon
    const usuarioId = pago.metadata?.usuario_id
    if (cuponUsado?.codigo && usuarioId) {
      await reservarUsoCuenta(usuarioId, cuponUsado.codigo, paymentId)
    }
  }

  // Enviar emails solo si el pago quedó aprobado Y es la primera vez que lo procesamos
  if (pago.status === 'approved' && !yaEstabaAprobado) {
    const pedidoGuardado = {
      cliente_nombre: datosCliente.nombre || null,
      comprador_email: datosCliente.email || pago.payer?.email || null,
      cliente_telefono: datosCliente.telefono || null,
      cliente_dni: datosCliente.dni || null,
      items,
      total: pago.transaction_amount,
      zona_envio: zonaEnvio?.nombre || null,
      costo_envio: zonaEnvio?.costo || 0,
      direccion: datosCliente.calle
        ? {
            calle: datosCliente.calle,
            numero: datosCliente.numero,
            pisoDepto: datosCliente.pisoDepto || datosCliente.piso_depto || null,
            ciudad: datosCliente.ciudad,
            codigoPostal: datosCliente.codigoPostal || datosCliente.codigo_postal || null,
          }
        : null,
      notas: datosCliente.notas || null,
      mercadopago_payment_id: String(paymentId),
    }

    let facturaInfo = null
    try {
      facturaInfo = await generarFactura({ ...pedidoGuardado, factura_cae: null })
      if (!facturaInfo.yaExistia) {
        await supabaseAdmin
          .from('pedidos')
          .update({
            factura_cae: facturaInfo.cae,
            factura_vencimiento_cae: facturaInfo.vencimiento,
            factura_numero: facturaInfo.numero,
            factura_pdf_url: facturaInfo.pdfUrl,
          })
          .eq('mercadopago_payment_id', String(paymentId))
      }
    } catch (error) {
      console.error('Error al generar factura automática:', error)
    }

    await enviarEmailConfirmacionCliente({ ...pedidoGuardado, factura_pdf_url: facturaInfo?.pdfUrl || null })
    await enviarEmailNotificacionVenta(pedidoGuardado)
  }

  return pago
}

async function facturarPedido(pedido) {

      .from('pedidos')
      .update({

        factura_vencimiento_cae: facturaInfo.vencimiento,
        factura_numero: facturaInfo.numero,
        factura_pdf_url: facturaInfo.pdfUrl,
      })
      .eq('id', pedido.id)

    if (error) {
      console.error('La factura se emitió pero no se pudo guardar en el pedido:', error, facturaInfo)
    }
    return facturaInfo
  } catch (error) {
    console.error('Error al generar la factura de transferencia:', error)
    return null
  }
}

// Marca el pedido como pagado, factura por el total con descuento y avisa al cliente y al vendedor.
export async function confirmarPagoTransferencia(pedidoId) {
  const { data: pedido, error } = await supabaseAdmin
    .from('pedidos')
    .select('*')
    .eq('id', pedidoId)
    .maybeSingle()

  if (error) throw error
  if (!pedido) return { ok: false, error: 'No existe ese pedido' }
  if (pedido.metodo_pago !== 'transferencia') {
    return { ok: false, error: 'Ese pedido no es por transferencia' }
  }

  // Ya estaba pagado: solo reintenta la factura si había fallado, sin mandar otro email
  if (pedido.estado === 'approved') {
    if (pedido.factura_cae) return { ok: true, yaConfirmado: true, facturaOk: true }
    const factura = await facturarPedido(pedido)
    return { ok: true, yaConfirmado: true, facturaOk: Boolean(factura) }
  }

  // El cambio de estado solo se aplica si sigue pendiente: si el botón se toca dos veces
  // a la vez, una sola pasa y no se factura ni se manda el mail por duplicado
  const { data: actualizados, error: errorEstado } = await supabaseAdmin
    .from('pedidos')
    .update({ estado: 'approved' })
    .eq('id', pedidoId)
    .eq('estado', 'pending')
    .select('id')

  if (errorEstado) throw errorEstado
  if (!actualizados || actualizados.length === 0) {
    return { ok: true, yaConfirmado: true, facturaOk: Boolean(pedido.factura_cae) }
  }

  const factura = await facturarPedido(pedido)

  const pedidoPagado = {
    ...pedido,
    estado: 'approved',
    factura_pdf_url: factura?.pdfUrl || null,
  }

  // Al cliente: confirmación de pago
  await enviarEmailConfirmacionCliente(pedidoPagado)

  // Al vendedor: aviso de venta confirmada (igual que con Mercado Pago)
  await enviarEmailNotificacionVenta(pedidoPagado)

  return { ok: true, yaConfirmado: false, facturaOk: Boolean(factura) }
}
